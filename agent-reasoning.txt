# -------------------------------------------------------------------------
ollama_query() {
    local prompt="$1"
    local model="${2:-$OLLAMA_MODEL}"
    local options="${3:-}"
    local stream="${4:-false}"
    
    local payload
    payload=$(jq -n \
        --arg model "$model" \
        --arg prompt "$prompt" \
        --argjson stream "$stream" \
        --argjson options "$options" \
        '{
            model: $model,
            prompt: $prompt,
            stream: $stream,
            options: ($options // {})
        }')
    
    local response error_msg http_code
    local timeout=30
    
    # Try Ollama API
    response=$(curl -s \
        --max-time "$timeout" \
        --connect-timeout 10 \
        -H "Content-Type: application/json" \
        -d "$payload" \
        "${OLLAMA_HOST}/api/generate" 2>&1) || true
    
    http_code=$(echo "$response" | grep -o 'HTTP/[0-9.]* [0-9]*' | tail -1 | awk '{print $2}')
    
    if [[ -n "$http_code" && "$http_code" -eq 200 ]]; then
        echo "$response" | jq -r '.response // empty'
        return 0
    else
        # Analyze error type
        if [[ "$response" == *"Connection refused"* ]] || [[ "$response" == *"Failed to connect"* ]]; then
            error_msg="Ollama connection refused - service may not be running"
        elif [[ "$response" == *"timed out"* ]]; then
            error_msg="Ollama request timed out after ${timeout}s"
        elif [[ "$response" == *"model not found"* ]]; then
            error_msg="Model '$model' not found in Ollama"
        elif [[ -n "$http_code" ]]; then
            error_msg="Ollama HTTP error: $http_code"
        else
            error_msg="Unknown Ollama error"
        fi
        
        log_message "ERROR" "Ollama query failed: $error_msg" "API_BRIDGE"
        echo "__OLLAMA_ERROR__:$error_msg"
        return 1
    fi
}

# -------------------------------------------------------------------------
#   3️⃣  Gemini Fallback
# -------------------------------------------------------------------------
gemini_fallback() {
    local prompt="$1"
    local model="${2:-gemini-pro}"
    
    if [[ -z "$GEMINI_API_KEY" ]]; then
        log_message "ERROR" "No Gemini API key available for fallback" "FALLBACK"
        echo "__FALLBACK_ERROR__:No API keys available"
        return 1
    fi
    
    local payload
    payload=$(jq -n \
        --arg model "$model" \
        --arg prompt "$prompt" \
        '{
            contents: [{
                parts: [{
                    text: $prompt
                }]
            }]
        }')
    
    local response
    response=$(curl -s \
        -H "Content-Type: application/json" \
        -H "x-goog-api-key: $GEMINI_API_KEY" \
        -d "$payload" \
        "https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent" 2>&1)
    
    if echo "$response" | jq -e '.candidates[0].content.parts[0].text' >/dev/null 2>&1; then
        echo "$response" | jq -r '.candidates[0].content.parts[0].text'
        log_message "INFO" "Successfully fell back to Gemini API" "FALLBACK"
        return 0
    else
        log_message "ERROR" "Gemini fallback also failed" "FALLBACK"
        echo "__FALLBACK_ERROR__:Gemini API request failed"
        return 1
    fi
}

# -------------------------------------------------------------------------
#   4️⃣  Smart API Router with Fallback
# -------------------------------------------------------------------------
query_ai() {
    local prompt="$1"
    local model="${2:-}"
    local options="${3:-}"
    local force_fallback="${4:-false}"
    
    # Default options if not provided
    if [[ -z "$options" ]]; then
        options='{"temperature": 0.7, "num_predict": 512}'
    fi
    
    # Check if we should force fallback
    if [[ "$force_fallback" == "true" ]]; then
        log_message "WARN" "Forcing fallback to Gemini (bypassing Ollama)" "API_ROUTER"
        gemini_fallback "$prompt" "${model:-gemini-pro}"
        return $?
    fi
    
    # Try Ollama first
    local ollama_response
    if ollama_response=$(ollama_query "$prompt" "${model:-$OLLAMA_MODEL}" "$options"); then
        if [[ "$ollama_response" != "__OLLAMA_ERROR__"* ]]; then
            echo "$ollama_response"
            return 0
        fi
    fi
    
    # Extract error message if present
    local error_msg="${ollama_response#__OLLAMA_ERROR__:}"
    
    # Fallback to Gemini
    log_message "WARN" "Falling back to Gemini API. Reason: $error_msg" "API_ROUTER"
    
    # Enhance prompt with context about the fallback
    local enhanced_prompt
    enhanced_prompt="[Context: This query was attempted on local Ollama but failed due to: $error_msg]\n\nOriginal query: $prompt"
    
    gemini_fallback "$enhanced_prompt" "gemini-pro"
}

agent_communicate() {
    local sender="$1"
    local receiver="$2"
    local message="$3"
    local context="${4:-}"
    
    # Check if communication is allowed
    if [[ ! " ${AGENT_COMMUNICATION[$receiver]} " =~ " $sender " ]]; then
        log_message "WARN" "Agent $sender cannot communicate directly with $receiver" "COMMS"
        return 1
    fi
    
    # Format the communication
    local formatted_message
    formatted_message=$(cat <<EOF
[AGENT_COMMUNICATION]
FROM: $sender
TO: $receiver
CONTEXT: $context
MESSAGE: $message
EOF
)
    
    log_message "AGENT_COLLAB" "$sender → $receiver: ${message:0:100}..." "$sender"
    
    # Store communication in shared memory
    sql_exec <<EOF
INSERT INTO agent_communications 
    (sender, receiver, message, context, timestamp)
VALUES 
    ('$sender', '$receiver', '$(escape_sql "$message")', '$(escape_sql "$context")', CURRENT_TIMESTAMP);
EOF
    
    echo "$formatted_message"
}

execute_agent_with_context() {
    local agent_name="$1"
    local prompt="$2"
    local context="${3:-}"
    local previous_agents_results="${4:-}"
    
    local agent_role="${AGENT_ROLES[$agent_name]}"
    
    # Build enhanced prompt with context
    local enhanced_prompt
    enhanced_prompt=$(cat <<EOF
[ROLE: ${agent_role}]
[AGENT: ${agent_name}]

CONTEXT FROM PREVIOUS ANALYSIS:
${previous_agents_results}

SPECIFIC INSTRUCTIONS:
${context}

TASK TO EXECUTE:
${prompt}

YOUR ANALYSIS:
EOF
)
    
    # Check if this is CODE agent receiving from SIGN/LOOP
    if [[ "$agent_name" == "CODE" && -n "$context" ]]; then
        enhanced_prompt+="\n\n[NOTE: You are receiving interpreted symbols from SIGN and optimization patterns from LOOP. Please translate these insights into executable code.]"
    fi
    
    # Query AI
    local response
    response=$(query_ai "$enhanced_prompt" "$OLLAMA_MODEL" '{"temperature": 0.5, "num_predict": 1024}')
    
    if [[ "$response" == "__FALLBACK_ERROR__"* ]]; then
        log_message "ERROR" "All AI services failed for agent $agent_name" "$agent_name"
        response="[ERROR: AI service unavailable. Agent $agent_name cannot process.]"
    fi
    
    # Store result
    sql_exec <<EOF
INSERT INTO agent_executions 
    (agent_name, prompt, context, response, timestamp)
VALUES 
    ('$agent_name', '$(escape_sql "$prompt")', '$(escape_sql "$context")', '$(escape_sql "$response")', CURRENT_TIMESTAMP);
EOF
    
    echo "$response"
}

collaborative_analysis() {
    local main_prompt="$1"
    local focus_agents="${2:-ALL}"
    
    log_message "INFO" "Starting collaborative analysis with focus: $focus_agents" "SYNC"
    
    # Phase 1: Individual Analysis
    local analysis_results=()
    local analysis_agents=()
    
    if [[ "$focus_agents" == "ALL" ]]; then
        analysis_agents=("CUBE" "CORE" "LOOP" "WAVE" "SIGN" "LINE" "COIN")
    else
        IFS=' ' read -ra analysis_agents <<< "$focus_agents"
    fi
    
    for agent in "${analysis_agents[@]}"; do
        log_message "INFO" "Phase 1: $agent analyzing..." "SYNC"
        
        local agent_prompt
        case "$agent" in
            SIGN)
                agent_prompt="Interpret symbols and hidden meanings in: $main_prompt"
                ;;
            LOOP)
                agent_prompt="Identify patterns and optimization opportunities in: $main_prompt"
                ;;
            CUBE)
                agent_prompt="Analyze structure and components of: $main_prompt"
                ;;
            *)
                agent_prompt="Analyze from your role perspective: $main_prompt"
                ;;
        esac
        
        local result
        result=$(execute_agent_with_context "$agent" "$agent_prompt" "" "")
        analysis_results+=("$agent:$result")
        
        log_message "INFO" "$agent analysis complete" "SYNC"
    done
    
    # Phase 2: Agent Communication & Synthesis
    local synthesized_context=""
    for result_entry in "${analysis_results[@]}"; do
        local agent="${result_entry%%:*}"
        local result="${result_entry#*:}"
        synthesized_context+="=== $agent ===\n$result\n\n"
    done
    
    # Phase 3: Specialized Agent Execution (CODE, WORK, etc.)
    local final_results=()
    
    # Check if we need CODE agent
    if [[ "$focus_agents" == "ALL" ]] || [[ " $focus_agents " =~ " CODE " ]]; then
        log_message "INFO" "Phase 3: CODE agent executing with insights..." "SYNC"
        
        # Prepare communications from SIGN and LOOP to CODE
        local sign_to_code loop_to_code
        
        for result_entry in "${analysis_results[@]}"; do
            local agent="${result_entry%%:*}"
            local result="${result_entry#*:}"
            
            case "$agent" in
                SIGN)
                    sign_to_code=$(agent_communicate "SIGN" "CODE" "$result" "Symbol interpretation for code generation")
                    ;;
                LOOP)
                    loop_to_code=$(agent_communicate "LOOP" "CODE" "$result" "Optimization patterns for implementation")
                    ;;
            esac
        done
        
        # CODE agent execution with received communications
        local code_context="Received insights:\n"
        [[ -n "$sign_to_code" ]] && code_context+="- From SIGN: ${sign_to_code}\n"
        [[ -n "$loop_to_code" ]] && code_context+="- From LOOP: ${loop_to_code}\n"
        
        local code_result
        code_result=$(execute_agent_with_context "CODE" \
            "Generate executable code based on all analysis insights" \
            "$code_context" \
            "$synthesized_context")
        
        final_results+=("CODE:$code_result")
    fi
    
    # Check if we need WORK agent
    if [[ "$focus_agents" == "ALL" ]] || [[ " $focus_agents " =~ " WORK " ]]; then
        log_message "INFO" "Phase 3: WORK agent executing with insights..." "SYNC"
        
        local work_result
        work_result=$(execute_agent_with_context "WORK" \
            "Create actionable steps and implementation plan" \
            "Based on comprehensive analysis from all agents" \
            "$synthesized_context")
        
        final_results+=("WORK:$work_result")
    fi
    
    # Phase 4: Final Synchronization
    log_message "INFO" "Phase 4: Final synchronization..." "SYNC"
    
    local sync_prompt="Synthesize final results from all agents:\n\n"
    sync_prompt+="Original Task: $main_prompt\n\n"
    sync_prompt+="Agent Analyses:\n$synthesized_context\n"
    
    if [[ ${#final_results[@]} -gt 0 ]]; then
        sync_prompt+="\nSpecialized Executions:\n"
        for result_entry in "${final_results[@]}"; do
            local agent="${result_entry%%:*}"
            local result="${result_entry#*:}"
            sync_prompt+="=== $agent ===\n$result\n\n"
        done
    fi
    
    sync_prompt+="Provide a comprehensive final answer integrating all insights."
    
    local final_output
    final_output=$(execute_agent_with_context "SYNC" "$sync_prompt" "" "")
    
    # Output structured results
    jq -n \
        --arg prompt "$main_prompt" \
        --argjson analysis_agents "$(printf '%s\n' "${analysis_agents[@]}" | jq -R . | jq -s .)" \
        --argjson analysis_results "$(printf '%s\n' "${analysis_results[@]}" | jq -R . | jq -s .)" \
        --argjson final_results "$(printf '%s\n' "${final_results[@]}" | jq -R . | jq -s .)" \
        --arg final_output "$final_output" \
        '{
            status: "success",
            original_prompt: $prompt,
            analysis_phase: {
                agents: $analysis_agents,
                results: $analysis_results
            },
            execution_phase: {
                results: $final_results
            },
            final_synthesis: $final_output,
            timestamp: now | todate
        }'
}

discover_ollama_models() {
    local timeout=5
    
    log_message "INFO" "Discovering Ollama models..." "MODEL_MGMT"
    
    local response
    response=$(curl -s --max-time "$timeout" "${OLLAMA_HOST}/api/tags" 2>/dev/null || true)
    
    if [[ -n "$response" ]] && echo "$response" | jq -e '.models' >/dev/null 2>&1; then
        local models
        models=$(echo "$response" | jq -r '.models[].name')
        
        if [[ -z "$models" ]]; then
            log_message "WARN" "Ollama running but no models found" "MODEL_MGMT"
            echo ""
        else
            log_message "INFO" "Found models: $(echo "$models" | tr '\n' ', ')" "MODEL_MGMT"
            echo "$models"
        fi
    else
        log_message "ERROR" "Failed to discover Ollama models" "MODEL_MGMT"
        echo ""
    fi
}

get_active_model() {
    if [[ -f "$CONFIG_FILE" ]]; then
        jq -r '.active_model // empty' "$CONFIG_FILE" 2>/dev/null || echo "$OLLAMA_MODEL"
    else
        echo "$OLLAMA_MODEL"
    fi
}

set_active_model() {
    local model="$1"
    
    # Verify model exists if Ollama is available
    local available_models
    available_models=$(discover_ollama_models)
    
    if [[ -n "$available_models" ]] && ! echo "$available_models" | grep -q "^${model}$"; then
        log_message "WARN" "Model '$model' not found in Ollama, but setting anyway" "MODEL_MGMT"
    fi
    
    # Update config
    jq -n --arg model "$model" '{active_model: $model}' > "$CONFIG_FILE"
    log_message "INFO" "Active model set to: $model" "MODEL_MGMT"
}

CREATE INDEX IF NOT EXISTS idx_agent_comms ON agent_communications (sender, receiver, timestamp);

COMMIT;

import axios, { AxiosError, AxiosResponse } from 'axios';
import { config } from '../config';

// Types
interface OllamaOptions {
  temperature?: number;
  num_predict?: number;
  top_p?: number;
  repeat_penalty?: number;
  seed?: number;
  [key: string]: any;
}

interface OllamaRequest {
  model: string;
  prompt: string;
  stream?: boolean;
  options?: OllamaOptions;
}

interface OllamaResponse {
  model: string;
  response: string;
  done: boolean;
  context?: number[];
  total_duration?: number;
  load_duration?: number;
  prompt_eval_count?: number;
  eval_count?: number;
}

interface AgentResponse {
  agent: string;
  decision: string;
  reasoning: string;
  confidence: number;
  rawResponse: string;
}

interface OllamaError {
  type: 'connection' | 'timeout' | 'model_not_found' | 'api_error' | 'rate_limit' | 'parsing';
  message: string;
  details?: any;
}

// Configuration
const OLLAMA_HOST = process.env.OLLAMA_HOST || 'http://localhost:11434';
const OLLAMA_TIMEOUT = parseInt(process.env.OLLAMA_TIMEOUT || '30000');
const OLLAMA_RETRY_ATTEMPTS = parseInt(process.env.OLLAMA_RETRY_ATTEMPTS || '2');
const GEMINI_API_KEY = process.env.GEMINI_API_KEY || config.geminiApiKey;
const FALLBACK_ENABLED = process.env.FALLBACK_ENABLED !== 'false';

// Model cache
let availableModels: string[] = [];
let lastModelDiscovery = 0;
const MODEL_CACHE_TTL = 5 * 60 * 1000; // 5 minutes

/**
 * Enhanced Ollama API client with robust error handling
 */
export class OllamaBridge {
  private static instance: OllamaBridge;
  
  static getInstance(): OllamaBridge {
    if (!OllamaBridge.instance) {
      OllamaBridge.instance = new OllamaBridge();
    }
    return OllamaBridge.instance;
  }

  /**
   * Query Ollama with enhanced error handling and fallback
   */
  async queryOllama(
    prompt: string, 
    model: string = 'llama3', 
    options: OllamaOptions = {}, 
    stream: boolean = false
  ): Promise<OllamaResponse> {
    const requestData: OllamaRequest = {
      model,
      prompt,
      stream,
      options: {
        temperature: 0.7,
        num_predict: 512,
        ...options
      }
    };

    // Log the attempt
    console.log(`[OllamaBridge] Querying model ${model} with ${prompt.length} chars`);

    try {
      const response = await this.makeRequestWithRetry(
        `${OLLAMA_HOST}/api/generate`,
        requestData,
        OLLAMA_RETRY_ATTEMPTS
      );

      return this.parseResponse(response, model, stream);

    } catch (error) {
      const ollamaError = this.classifyError(error);
      
      // Enhanced error logging
      console.error(`[OllamaBridge] Query failed:`, {
        errorType: ollamaError.type,
        message: ollamaError.message,
        model,
        promptLength: prompt.length,
        timestamp: new Date().toISOString()
      });

      // If fallback is enabled and this is a recoverable error
      if (FALLBACK_ENABLED && this.isRecoverableError(ollamaError)) {
        console.warn(`[OllamaBridge] Attempting fallback to Gemini API`);
        return await this.fallbackToGemini(prompt, model, ollamaError);
      }

      throw new Error(`Ollama query failed: ${ollamaError.message}`);
    }
  }

  /**
   * Make HTTP request with retry logic
   */
  private async makeRequestWithRetry(
    url: string,
    data: OllamaRequest,
    retries: number
  ): Promise<AxiosResponse> {
    for (let attempt = 0; attempt <= retries; attempt++) {
      try {
        const response = await axios.post(url, data, {
          timeout: OLLAMA_TIMEOUT,
          headers: {
            'Content-Type': 'application/json'
          },
          validateStatus: (status) => status === 200
        });

        return response;

      } catch (error) {
        const axiosError = error as AxiosError;
        
        // Log retry attempts
        if (attempt < retries) {
          const delay = Math.pow(2, attempt) * 1000; // Exponential backoff
          console.warn(`[OllamaBridge] Attempt ${attempt + 1} failed, retrying in ${delay}ms:`, 
            axiosError.message);
          await new Promise(resolve => setTimeout(resolve, delay));
          continue;
        }

        // Last attempt failed
        throw error;
      }
    }

    throw new Error('Max retry attempts exceeded');
  }

  /**
   * Classify Ollama errors for better handling
   */
  private classifyError(error: any): OllamaError {
    const axiosError = error as AxiosError;
    
    if (axiosError.code === 'ECONNREFUSED') {
      return {
        type: 'connection',
        message: 'Cannot connect to Ollama. Ensure it is running at the correct host/port.',
        details: { host: OLLAMA_HOST }
      };
    }

    if (axiosError.code === 'ETIMEDOUT') {
      return {
        type: 'timeout',
        message: `Ollama request timed out after ${OLLAMA_TIMEOUT}ms.`,
        details: { timeout: OLLAMA_TIMEOUT }
      };
    }

    if (axiosError.response?.status === 404) {
      return {
        type: 'model_not_found',
        message: `Model not found. Please ensure the model is pulled: 'ollama pull <model-name>'`,
        details: { status: 404 }
      };
    }

    if (axiosError.response?.status === 429) {
      return {
        type: 'rate_limit',
        message: 'Ollama rate limit exceeded. Please try again later.',
        details: { status: 429 }
      };
    }

    // Try to extract error from response body
    if (axiosError.response?.data) {
      try {
        const errorData = typeof axiosError.response.data === 'string' 
          ? JSON.parse(axiosError.response.data)
          : axiosError.response.data;
        
        return {
          type: 'api_error',
          message: errorData.error || 'Ollama API error',
          details: errorData
        };
      } catch {
        // If we can't parse, return generic error
      }
    }

    // Default parsing error
    if (error.message?.includes('Unexpected token') || error.message?.includes('JSON')) {
      return {
        type: 'parsing',
        message: 'Failed to parse Ollama response. The response format may have changed.',
        details: { rawResponse: error.message }
      };
    }

    return {
      type: 'api_error',
      message: `Ollama error: ${error.message || 'Unknown error'}`,
      details: error
    };
  }

  /**
   * Check if error is recoverable (should trigger fallback)
   */
  private isRecoverableError(error: OllamaError): boolean {
    const recoverableTypes: OllamaError['type'][] = [
      'connection',
      'timeout',
      'model_not_found',
      'api_error'
    ];
    
    return recoverableTypes.includes(error.type) && !!GEMINI_API_KEY;
  }

  /**
   * Fallback to Gemini API
   */
  private async fallbackToGemini(
    prompt: string,
    originalModel: string,
    ollamaError: OllamaError
  ): Promise<OllamaResponse> {
    try {
      console.log(`[OllamaBridge] Starting fallback to Gemini API. Original error: ${ollamaError.message}`);
      
      const fallbackPrompt = this.createFallbackPrompt(prompt, originalModel, ollamaError);
      
      const response = await axios.post(
        `https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent`,
        {
          contents: [{
            parts: [{ text: fallbackPrompt }]
          }]
        },
        {
          headers: {
            'Content-Type': 'application/json',
            'x-goog-api-key': GEMINI_API_KEY
          },
          timeout: 30000
        }
      );

      const geminiText = response.data?.candidates?.[0]?.content?.parts?.[0]?.text;
      
      if (!geminiText) {
        throw new Error('Gemini API returned empty response');
      }

      // Log successful fallback
      console.log(`[OllamaBridge] Fallback to Gemini successful. Response length: ${geminiText.length} chars`);
      
      // Return formatted as Ollama response for compatibility
      return {
        model: 'gemini-pro (fallback)',
        response: geminiText,
        done: true,
        context: []
      };

    } catch (fallbackError) {
      console.error(`[OllamaBridge] Gemini fallback also failed:`, fallbackError);
      throw new Error(`Both Ollama and Gemini failed: ${ollamaError.message}`);
    }
  }

  /**
   * Create enhanced prompt for fallback
   */
  private createFallbackPrompt(
    originalPrompt: string,
    originalModel: string,
    error: OllamaError
  ): string {
    return `
[Context: This query was originally intended for local Ollama model "${originalModel}" 
but failed due to: ${error.message}]

[Instructions: Please provide a response as if you were the local AI model. 
Maintain the expected response format and tone.]

Original query: ${originalPrompt}
`;
  }

  /**
   * Parse Ollama response with robust handling
   */
  private parseResponse(
    response: AxiosResponse,
    model: string,
    stream: boolean
  ): OllamaResponse {
    try {
      if (stream) {
        return this.parseStreamResponse(response.data, model);
      } else {
        return this.parseStandardResponse(response.data, model);
      }
    } catch (error) {
      console.error(`[OllamaBridge] Failed to parse response:`, error);
      
      // Attempt to extract meaningful information even if parsing fails
      const rawData = response.data;
      let extractedResponse = '';
      
      if (typeof rawData === 'string') {
        // Try to find JSON-like structure
        const jsonMatch = rawData.match(/"response":\s*"([^"]+)"/);
        if (jsonMatch) {
          extractedResponse = jsonMatch[1];
        } else {
          // Try to extract any text that looks like a response
          extractedResponse = rawData.substring(0, 500);
        }
      } else if (rawData && typeof rawData === 'object') {
        // Try various known response formats
        extractedResponse = 
          rawData.response || 
          rawData.text || 
          rawData.content ||
          JSON.stringify(rawData).substring(0, 500);
      }
      
      return {
        model,
        response: extractedResponse || 'Unable to parse response',
        done: true,
        context: []
      };
    }
  }

  /**
   * Parse standard (non-streaming) response
   */
  private parseStandardResponse(data: any, model: string): OllamaResponse {
    // Handle various response formats
    let responseText = '';
    let done = true;
    
    if (typeof data === 'string') {
      // Try to parse as JSON
      try {
        const parsed = JSON.parse(data);
        responseText = parsed.response || parsed.text || '';
        done = parsed.done !== false;
      } catch {
        // If not JSON, use as-is
        responseText = data;
      }
    } else if (data && typeof data === 'object') {
      // Handle different possible structures
      if (data.response !== undefined) {
        responseText = data.response;
        done = data.done !== false;
      } else if (data.text !== undefined) {
        responseText = data.text;
      } else if (data.content !== undefined) {
        responseText = data.content;
      } else {
        // Last resort: stringify
        responseText = JSON.stringify(data);
      }
    }
    
    return {
      model,
      response: responseText,
      done,
      context: data.context || [],
      total_duration: data.total_duration,
      load_duration: data.load_duration,
      prompt_eval_count: data.prompt_eval_count,
      eval_count: data.eval_count
    };
  }

  /**
   * Parse streaming response
   */
  private parseStreamResponse(data: any, model: string): OllamaResponse {
    // Streaming responses come as newline-delimited JSON
    const lines = data.split('\n').filter((line: string) => line.trim());
    let fullResponse = '';
    let context: number[] = [];
    
    for (const line of lines) {
      try {
        const parsed = JSON.parse(line);
        if (parsed.response) {
          fullResponse += parsed.response;
        }
        if (parsed.done && parsed.context) {
          context = parsed.context;
        }
      } catch {
        // Skip invalid JSON lines
      }
    }
    
    return {
      model,
      response: fullResponse,
      done: true,
      context
    };
  }

  /**
   * Discover available Ollama models
   */
  async discoverModels(forceRefresh: boolean = false): Promise<string[]> {
    const now = Date.now();
    
    // Use cache if valid
    if (!forceRefresh && availableModels.length > 0 && 
        (now - lastModelDiscovery) < MODEL_CACHE_TTL) {
      return availableModels;
    }
    
    try {
      const response = await axios.get(`${OLLAMA_HOST}/api/tags`, {
        timeout: 5000
      });
      
      if (response.data?.models) {
        availableModels = response.data.models.map((model: any) => model.name);
        lastModelDiscovery = now;
        console.log(`[OllamaBridge] Discovered ${availableModels.length} models`);
        return availableModels;
      }
      
      return [];
    } catch (error) {
      console.error(`[OllamaBridge] Failed to discover models:`, error);
      return [];
    }
  }

  /**
   * Test Ollama connection and model
   */
  async testConnection(model: string = 'llama3'): Promise<{
    success: boolean;
    message: string;
    details?: any;
  }> {
    try {
      // Test connectivity
      await axios.get(`${OLLAMA_HOST}/api/tags`, { timeout: 3000 });
      
      // Test model
      const response = await this.queryOllama(
        'Please respond with the word "READY" if you are operational.',
        model,
        { temperature: 0.1, num_predict: 10 },
        false
      );
      
      const isReady = response.response?.toUpperCase().includes('READY');
      
      return {
        success: isReady,
        message: isReady 
          ? `Model ${model} is ready and responding` 
          : `Model responded but not as expected: ${response.response}`,
        details: {
          model,
          response: response.response,
          responseTime: response.total_duration
        }
      };
      
    } catch (error) {
      const classifiedError = this.classifyError(error);
      return {
        success: false,
        message: `Connection test failed: ${classifiedError.message}`,
        details: classifiedError
      };
    }
  }
}

/**
 * Enhanced agent response parser
 */
export class AgentResponseParser {
  
  /**
   * Parse agent response with multiple fallback strategies
   */
  static parseAgentResponse(
    rawResponse: string, 
    expectedAgent: string
  ): AgentResponse {
    const strategies = [
      this.parseStructuredJson,
      this.parseMarkdownWithHeaders,
      this.parseKeyValueFormat,
      this.parseNaturalLanguage
    ];
    
    for (const strategy of strategies) {
      try {
        const parsed = strategy(rawResponse, expectedAgent);
        if (parsed && parsed.decision && parsed.reasoning) {
          return {
            ...parsed,
            rawResponse,
            confidence: parsed.confidence || this.calculateConfidence(parsed)
          };
        }
      } catch {
        // Try next strategy
        continue;
      }
    }
    
    // Last resort fallback
    return this.createFallbackResponse(rawResponse, expectedAgent);
  }
  
  /**
   * Strategy 1: Parse structured JSON
   */
  private static parseStructuredJson(
    rawResponse: string, 
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    // Try to find JSON in the response
    const jsonMatch = rawResponse.match(/\{[\s\S]*\}/);
    if (!jsonMatch) return null;
    
    try {
      const parsed = JSON.parse(jsonMatch[0]);
      
      // Handle various JSON structures
      return {
        agent: parsed.agent || parsed.role || expectedAgent,
        decision: parsed.decision || parsed.result || parsed.output || '',
        reasoning: parsed.reasoning || parsed.analysis || parsed.explanation || '',
        confidence: parsed.confidence || parsed.certainty || 0.5
      };
    } catch {
      return null;
    }
  }
  
  /**
   * Strategy 2: Parse markdown with headers
   */
  private static parseMarkdownWithHeaders(
    rawResponse: string,
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    const decisionMatch = rawResponse.match(/##? Decision[:\s]*\n*(.*?)(?=\n##? |$)/is);
    const reasoningMatch = rawResponse.match(/##? Reasoning[:\s]*\n*(.*?)(?=\n##? |$)/is);
    const agentMatch = rawResponse.match(/##? Agent[:\s]*\n*(\w+)/i);
    
    if (!decisionMatch && !reasoningMatch) return null;
    
    return {
      agent: agentMatch?.[1] || expectedAgent,
      decision: decisionMatch?.[1]?.trim() || '',
      reasoning: reasoningMatch?.[1]?.trim() || '',
      confidence: this.extractConfidenceFromText(rawResponse)
    };
  }
  
  /**
   * Strategy 3: Parse key-value format
   */
  private static parseKeyValueFormat(
    rawResponse: string,
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    const lines = rawResponse.split('\n');
    let decision = '';
    let reasoning = '';
    let agent = expectedAgent;
    
    for (const line of lines) {
      const lowerLine = line.toLowerCase();
      if (lowerLine.includes('decision:')) {
        decision = line.split(':').slice(1).join(':').trim();
      } else if (lowerLine.includes('reasoning:')) {
        reasoning = line.split(':').slice(1).join(':').trim();
      } else if (lowerLine.includes('agent:')) {
        agent = line.split(':').slice(1).join(':').trim();
      }
    }
    
    if (!decision && !reasoning) return null;
    
    return {
      agent,
      decision,
      reasoning,
      confidence: this.extractConfidenceFromText(rawResponse)
    };
  }
  
  /**
   * Strategy 4: Parse natural language (last resort)
   */
  private static parseNaturalLanguage(
    rawResponse: string,
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    // Extract first sentence as decision, rest as reasoning
    const sentences = rawResponse.split(/[.!?]+/).filter(s => s.trim());
    
    if (sentences.length === 0) return null;
    
    return {
      agent: expectedAgent,
      decision: sentences[0].trim(),
      reasoning: sentences.slice(1).join('. ').trim() || sentences[0].trim(),
      confidence: 0.3 // Low confidence for this method
    };
  }
  
  /**
   * Extract confidence from text using various patterns
   */
  private static extractConfidenceFromText(text: string): number {
    const patterns = [
      /confidence[:\s]*(\d+\.?\d*)/i,
      /certainty[:\s]*(\d+\.?\d*)/i,
      /(\d+)% (confidence|certainty)/i,
      /I am (\d+)% sure/i
    ];
    
    for (const pattern of patterns) {
      const match = text.match(pattern);
      if (match) {
        const value = parseFloat(match[1]);
        if (!isNaN(value)) {
          return value > 1 ? value / 100 : value;
        }
      }
    }
    
    // Default medium confidence
    return 0.5;
  }
  
  /**
   * Calculate confidence based on response characteristics
   */
  private static calculateConfidence(parsed: Partial<AgentResponse>): number {
    let confidence = 0.5;
    
    // Longer reasoning suggests more confidence
    if (parsed.reasoning && parsed.reasoning.length > 50) {
      confidence += 0.2;
    }
    
    // Clear decision suggests more confidence
    if (parsed.decision && parsed.decision.length > 10) {
      confidence += 0.2;
    }
    
    // Contains numbers or specific references
    if (parsed.decision?.match(/\d+/) || parsed.reasoning?.match(/\d+/)) {
      confidence += 0.1;
    }
    
    return Math.min(0.95, confidence);
  }
  
  /**
   * Create fallback response when all parsing fails
   */
  private static createFallbackResponse(
    rawResponse: string,
    expectedAgent: string
  ): AgentResponse {
    return {
      agent: expectedAgent,
      decision: 'Unable to parse specific decision',
      reasoning: `Raw response: ${rawResponse.substring(0, 200)}...`,
      confidence: 0.1,
      rawResponse
    };
  }
}

/**
 * Enhanced Nexus Engine with multi-agent collaboration
 */
export class NexusEngine {
  private ollamaBridge: OllamaBridge;
  private activeModel: string = 'llama3';
  
  constructor() {
    this.ollamaBridge = OllamaBridge.getInstance();
  }
  
  /**
   * Process prompt with multi-agent collaboration
   */
  async processPrompt(
    prompt: string,
    options: {
      model?: string;
      useAgents?: string[];
      enableStreaming?: boolean;
    } = {}
  ): Promise<{
    success: boolean;
    result: any;
    agentDecisions: AgentResponse[];
    fallbackUsed: boolean;
    errors: string[];
  }> {
    const startTime = Date.now();
    const errors: string[] = [];
    let fallbackUsed = false;
    
    // Set active model
    if (options.model) {
      this.activeModel = options.model;
    }
    
    // Determine which agents to use
    const agents = options.useAgents || ['CUBE', 'CORE', 'LOOP', 'WAVE', 'SIGN', 'LINE', 'COIN', 'WORK'];
    
    console.log(`[NexusEngine] Processing prompt with ${agents.length} agents`);
    
    try {
      // Phase 1: Individual agent analysis
      const agentPromises = agents.map(agent => 
        this.executeAgent(agent, prompt, options.enableStreaming)
      );
      
      const agentResults = await Promise.allSettled(agentPromises);
      
      // Process agent results
      const successfulAgents: AgentResponse[] = [];
      const agentDecisions: AgentResponse[] = [];
      
      agentResults.forEach((result, index) => {
        const agent = agents[index];
        
        if (result.status === 'fulfilled') {
          const agentResponse = AgentResponseParser.parseAgentResponse(
            result.value,
            agent
          );
          
          successfulAgents.push(agentResponse);
          agentDecisions.push(agentResponse);
          
          console.log(`[NexusEngine] Agent ${agent} completed with confidence ${agentResponse.confidence}`);
        } else {
          errors.push(`Agent ${agent} failed: ${result.reason}`);
          console.error(`[NexusEngine] Agent ${agent} failed:`, result.reason);
        }
      });
      
      // Phase 2: Agent collaboration (if we have multiple successful agents)
      let collaborationResult = '';
      
      if (successfulAgents.length > 1) {
        collaborationResult = await this.coordinateAgents(
          prompt,
          successfulAgents,
          options.enableStreaming
        );
      } else if (successfulAgents.length === 1) {
        collaborationResult = successfulAgents[0].decision;
      } else {
        throw new Error('All agents failed to process the prompt');
      }
      
      // Phase 3: Final synthesis
      const finalResult = await this.synthesizeResult(
        prompt,
        collaborationResult,
        agentDecisions,
        options.enableStreaming
      );
      
      const processingTime = Date.now() - startTime;
      
      console.log(`[NexusEngine] Processing completed in ${processingTime}ms`);
      console.log(`[NexusEngine] ${successfulAgents.length}/${agents.length} agents successful`);
      if (fallbackUsed) {
        console.warn(`[NexusEngine] Fallback mechanism was used`);
      }
      
      return {
        success: true,
        result: finalResult,
        agentDecisions,
        fallbackUsed,
        errors
      };
      
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : String(error);
      errors.push(`Processing failed: ${errorMessage}`);
      
      console.error(`[NexusEngine] Processing failed:`, error);
      
      // Final fallback: direct query without agents
      try {
        console.warn(`[NexusEngine] Attempting direct query as final fallback`);
        const directResponse = await this.ollamaBridge.queryOllama(
          prompt,
          this.activeModel,
          { temperature: 0.7 }
        );
        
        fallbackUsed = true;
        
        return {
          success: true,
          result: directResponse.response,
          agentDecisions: [],
          fallbackUsed: true,
          errors: [...errors, 'Used direct query fallback']
        };
        
      } catch (fallbackError) {
        return {
          success: false,
          result: 'All processing attempts failed',
          agentDecisions: [],
          fallbackUsed: false,
          errors: [...errors, `Fallback also failed: ${fallbackError}`]
        };
      }
    }
  }
  
  /**
   * Execute individual agent
   */
  private async executeAgent(
    agent: string,
    prompt: string,
    stream?: boolean
  ): Promise<string> {
    const agentPrompt = this.createAgentPrompt(agent, prompt);
    
    const response = await this.ollamaBridge.queryOllama(
      agentPrompt,
      this.activeModel,
      this.getAgentOptions(agent),
      stream
    );
    
    return response.response;
  }
  
  /**
   * Create agent-specific prompt
   */
  private createAgentPrompt(agent: string, prompt: string): string {
    const agentInstructions: Record<string, string> = {
      CUBE: 'Analyze the structure and components. Focus on architecture and organization.',
      CORE: 'Identify the core logic and truth. Focus on reasoning and principles.',
      LOOP: 'Look for patterns and optimization opportunities. Focus on iterations and improvements.',
      WAVE: 'Analyze flow, trends, and dynamics. Focus on changes and variability.',
      SIGN: 'Interpret symbols and hidden meanings. Focus on semiotics and interpretation.',
      LINE: 'Connect sequences and progressions. Focus on linear flow and connections.',
      COIN: 'Assess value, risk, and probability. Focus on cost-benefit analysis.',
      WORK: 'Create actionable steps and implementation plans. Focus on execution.',
      CODE: 'Generate executable code based on insights. Focus on practical implementation.',
      SYNC: 'Coordinate and synthesize multiple perspectives. Focus on integration.'
    };
    
    return `
[ROLE: ${agent}]
${agentInstructions[agent] || 'Analyze from your specialized perspective.'}

TASK: ${prompt}

Please provide:
1. Your analysis and decision
2. Your reasoning process
3. Any specific recommendations or next steps

Format your response clearly, using sections if helpful.
`;
  }
  
  /**
   * Get agent-specific options
   */
  private getAgentOptions(agent: string): OllamaOptions {
    const baseOptions: OllamaOptions = {
      temperature: 0.7,
      num_predict: 512
    };
    
    // Adjust based on agent type
    switch (agent) {
      case 'CUBE':
      case 'CORE':
        return { ...baseOptions, temperature: 0.3, num_predict: 768 };
      case 'LOOP':
      case 'CODE':
        return { ...baseOptions, temperature: 0.5, num_predict: 1024 };
      case 'SIGN':
      case 'WAVE':
        return { ...baseOptions, temperature: 0.8, num_predict: 512 };
      default:
        return baseOptions;
    }
  }
  
  /**
   * Coordinate multiple agents
   */
  private async coordinateAgents(
    originalPrompt: string,
    agentResults: AgentResponse[],
    stream?: boolean
  ): Promise<string> {
    const coordinationPrompt = this.createCoordinationPrompt(
      originalPrompt,
      agentResults
    );
    
    const response = await this.ollamaBridge.queryOllama(
      coordinationPrompt,
      this.activeModel,
      { temperature: 0.5, num_predict: 1024 },
      stream
    );
    
    return response.response;
  }
  
  /**
   * Create coordination prompt
   */
  private createCoordinationPrompt(
    originalPrompt: string,
    agentResults: AgentResponse[]
  ): string {
    let agentContributions = '';
    
    agentResults.forEach(result => {
      agentContributions += `
=== ${result.agent} (confidence: ${result.confidence.toFixed(2)}) ===
Decision: ${result.decision}
Reasoning: ${result.reasoning}

`;
    });
    
    return `
[ROLE: SYNCHRONIZER]
You are coordinating insights from multiple specialized agents.

ORIGINAL TASK: ${originalPrompt}

AGENT CONTRIBUTIONS:
${agentContributions}

YOUR TASK:
1. Synthesize the key insights from all agents
2. Identify areas of agreement and disagreement
3. Provide integrated recommendations
4. Highlight the most confident insights

Please provide a comprehensive coordination report.
`;
  }
  
  /**
   * Synthesize final result
   */
  private async synthesizeResult(
    originalPrompt: string,
    coordinationResult: string,
    agentDecisions: AgentResponse[],
    stream?: boolean
  ): Promise<string> {
    const synthesisPrompt = `
[ROLE: FINAL SYNTHESIZER]
Create the final output based on the coordinated analysis.

ORIGINAL REQUEST: ${originalPrompt}

COORDINATED ANALYSIS: ${coordinationResult}

AGENT DECISIONS SUMMARY:
${agentDecisions.map(a => `- ${a.agent}: ${a.decision.substring(0, 100)}...`).join('\n')}

Create a final response that:
1. Directly addresses the original request
2. Incorporates the coordinated insights
3. Is clear, actionable, and well-structured
4. Highlights key takeaways and next steps
`;
    
    const response = await this.ollamaBridge.queryOllama(
      synthesisPrompt,
      this.activeModel,
      { temperature: 0.4, num_predict: 768 },
      stream
    );
    
    return response.response;
  }
  
  /**
   * Get available models
   */
  async getAvailableModels(forceRefresh: boolean = false): Promise<string[]> {
    return this.ollamaBridge.discoverModels(forceRefresh);
  }
  
  /**
   * Set active model
   */
  setActiveModel(model: string): void {
    this.activeModel = model;
    console.log(`[NexusEngine] Active model set to: ${model}`);
  }
  
  /**
   * Get active model
   */
  getActiveModel(): string {
    return this.activeModel;
  }
  
  /**
   * Test Ollama connection
   */
  async testConnection(model?: string): Promise<{
    success: boolean;
    message: string;
    details?: any;
  }> {
    return this.ollamaBridge.testConnection(model || this.activeModel);
  }
}

              <div className="space-x-4">
                <button
                  type="submit"
                  disabled={loading || !prompt.trim()}
                  className="px-6 py-3 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed"
                >
                  {loading ? 'Processing...' : 'Analyze with Agents'}
                </button>
                
                <label className="inline-flex items-center space-x-2">
                  <input
                    type="checkbox"
                    checked={streaming}
                    onChange={(e) => setStreaming(e.target.checked)}
                    className="rounded"
                  />
                  <span className="text-sm text-gray-600">Enable streaming</span>
                </label>
              </div>

              <button
                type="button"
                onClick={() => setShowCollaboration(!showCollaboration)}
                className="px-4 py-2 text-sm bg-gray-200 rounded-lg hover:bg-gray-300"
              >
                {showCollaboration ? 'Hide Collaboration' : 'Show Collaboration'}
              </button>
            </div>
          </form>

          {/* Errors */}
          {errors.length > 0 && (
            <div className="p-4 bg-red-50 border border-red-200 rounded-lg">
              <h4 className="font-medium text-red-800 mb-2">Processing Errors:</h4>
              <ul className="space-y-1">
                {errors.map((error, index) => (
                  <li key={index} className="text-sm text-red-700">
                    • {error}
                  </li>
                ))}
              </ul>
              {fallbackUsed && (
                <div className="mt-3 p-3 bg-yellow-50 border border-yellow-200 rounded">
                  <div className="font-medium text-yellow-800">
                    ⚠️ Fallback to Gemini API was used
                  </div>
                  <div className="text-sm text-yellow-700 mt-1">
                    The system automatically switched to Gemini API after Ollama encountered issues.
                    This ensures your request was processed despite local model problems.
                  </div>
                </div>
              )}
            </div>
          )}

          {/* Response */}
          {response && (
            <div className="space-y-4">
              <div className="flex items-center justify-between">
                <h3 className="text-lg font-semibold">Analysis Result</h3>
                <div className="text-sm text-gray-500">
                  Generated by {fallbackUsed ? 'Gemini (fallback)' : activeModel}
                </div>
              </div>
              <div className="p-6 bg-white border rounded-lg whitespace-pre-wrap">
                {response}
              </div>
            </div>
          )}
        </div>

        {/* Right Panel */}
        <div className="space-y-6">
          <StatusSidebar />
          
          {showCollaboration && agentDecisions.length > 0 && (
            <div className="bg-white border rounded-lg p-6">
              <CollaborationVisualizer
                agentDecisions={agentDecisions}
                activeModel={activeModel}
              />
            </div>
          )}

          {/* Agent Communication Log */}
          {agentDecisions.length > 0 && (
            <div className="bg-white border rounded-lg p-4">
              <h4 className="font-medium mb-3">Agent Communication</h4>
              <div className="space-y-3 max-h-64 overflow-y-auto">
                {agentDecisions.map((agent) => (
                  <div key={agent.agent} className="p-3 bg-gray-50 rounded">
                    <div className="font-medium text-sm">{agent.agent}</div>
                    <div className="text-xs text-gray-600 mt-1">
                      Confidence: {Math.round(agent.confidence * 100)}%
                    </div>
                    <div className="text-sm mt-2">
                      {agent.decision.substring(0, 80)}...
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
};

#!/usr/bin/env bash
# startup.sh - Enhanced startup for NEXUS AI System

set -e

echo "🚀 Starting NEXUS AI Orchestration System v5.0"
echo "=============================================="

# Check for required dependencies
check_dependency() {
    if ! command -v "$1" &> /dev/null; then
        echo "❌ Missing dependency: $1"
        echo "   Please install $1 before continuing."
        return 1
    fi
    echo "✅ $1 found"
}

echo -e "\n🔍 Checking dependencies..."
check_dependency "node"
check_dependency "npm"
check_dependency "ollama" || {
    echo -e "\n⚠  Ollama not found. Installing..."
    curl -fsSL https://ollama.com/install.sh | sh
}

# Check if Ollama is running
echo -e "\n🔌 Checking Ollama service..."
if ! curl -s http://localhost:11434/api/tags >/dev/null 2>&1; then
    echo "🔄 Starting Ollama service..."
    ollama serve > /dev/null 2>&1 &
    OLLAMA_PID=$!
    sleep 5
    
    # Verify Ollama started
    if curl -s http://localhost:11434/api/tags >/dev/null 2>&1; then
        echo "✅ Ollama service started (PID: $OLLAMA_PID)"
    else
        echo "❌ Failed to start Ollama service"
        exit 1
    fi
else
    echo "✅ Ollama service is already running"
fi

# Check for models
echo -e "\n🤖 Checking Ollama models..."
MODELS=$(curl -s http://localhost:11434/api/tags | jq -r '.models[].name' 2>/dev/null || echo "")
if [[ -z "$MODELS" ]]; then
    echo "⚠  No models found. Pulling llama3..."
    ollama pull llama3
else
    echo "✅ Found models:"
    echo "$MODELS" | while read -r model; do
        echo "   • $model"
    done
fi

# Setup environment
echo -e "\n⚙️  Setting up environment..."
if [[ ! -f .env ]]; then
    cat > .env << EOF

# Agent Configuration
ENABLE_MULTI_AGENT=true
ENABLE_AGENT_COMMUNICATION=true
ENABLE_FALLBACK=true
ENABLE_STREAMING=false
EOF
    echo "✅ Created .env file"
else
    echo "✅ .env file already exists"
fi

# Install npm dependencies
echo -e "\n📦 Installing npm dependencies..."
npm install

echo -e "\n🗄️  Initializing database..."
node -e "
const { execSync } = require('child_process');

} catch (error) {
    console.error('Failed to create database directory:', error);
}

npm run build

npm start
