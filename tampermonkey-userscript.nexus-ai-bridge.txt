// ==UserScript==
// @name         NEXUS-2244-ULTIMATE
// @namespace    http://nexus.local
// @version      2244.1
// @description  NEXUS-2244: Visual Cognitive Engine, DOM-Aware, Time-Traveling AI
// @match        *://*/*
// @grant        GM_addStyle
// @grant        GM_xmlhttpRequest
// @grant        GM_download
// @grant        GM_setValue
// @grant        GM_getValue
// @connect      localhost
// @require      https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js
// ==/UserScript==

(function() {
    'use strict';

    const API = "http://localhost:3000";
    let lastDOMTree = [];

    // --- DOM ENGINE ---

    function captureDOMTree() {
        function walk(node, path = "0") {
            let out = [];
            if (!node || !node.tagName) return out;
            const id = node.id || "";
            const cls = node.className || "";
            const sig = node.tagName + "#" + id + "." + cls;
            out.push({ path, sig, rect: node.getBoundingClientRect() });
            Array.from(node.children).forEach((child, i) => {
                out.push(...walk(child, path + "." + i));
            });
            return out;
        }
        return walk(document.body);
    }

    function diffDOMTrees(before, after) {
        const mapA = new Map(before.map(n => [n.path, n.sig]));
        const mapB = new Map(after.map(n => [n.path, n.sig]));
        const changes = [];

        for (const [path, sig] of mapB.entries()) {
            if (!mapA.has(path)) {
                changes.push({ path, type: "added", sig });
            } else if (mapA.get(path) !== sig) {
                changes.push({ path, type: "modified", from: mapA.get(path), to: sig });
            }
        }
        for (const [path, sig] of mapA.entries()) {
            if (!mapB.has(path)) {
                changes.push({ path, type: "removed", sig });
            }
        }
        return changes;
    }

    function getNodeByPath(path) {
        try {
            let node = document.body;
            path.split(".").slice(1).forEach(i => {
                if(node && node.children) node = node.children[parseInt(i)];
            });
            return node;
        } catch { return null; }
    }

    function renderDOMHeatmap(domDiff) {
        let layer = document.getElementById("nexus-heatmap");
        if (!layer) {
            layer = document.createElement("div");
            layer.id = "nexus-heatmap";
            Object.assign(layer.style, {
                position: "fixed", left: "0", top: "0", width: "100vw", height: "100vh",
                pointerEvents: "none", zIndex: "999999", mixBlendMode: "screen"
            });
            document.body.appendChild(layer);
        }
        layer.innerHTML = "";

        domDiff.forEach(change => {
            const el = getNodeByPath(change.path);
            if (!el) return;
            const rect = el.getBoundingClientRect();
            const box = document.createElement("div");
            let color = "rgba(255,255,0,0.3)"; // Modified
            if (change.type === "added") color = "rgba(0,255,0,0.4)";
            if (change.type === "removed") color = "rgba(0,0,255,0.4)";

            Object.assign(box.style, {
                position: "absolute",
                left: rect.left + "px", top: rect.top + "px",
                width: rect.width + "px", height: rect.height + "px",
                background: color, border: "1px solid rgba(255,255,255,0.5)",
                borderRadius: "4px", animation: "nexus-pulse 2s infinite"
            });
            layer.appendChild(box);
        });
    }

    GM_addStyle(`@keyframes nexus-pulse { 0% { opacity: 0.2; } 50% { opacity: 0.8; } 100% { opacity: 0.2; } }`);

    // --- UI DASHBOARD ---

    function createUI() {
        if(document.getElementById("nexus-ui")) return;
        const div = document.createElement("div");
        div.id = "nexus-ui";
        Object.assign(div.style, {
            position: "fixed", bottom: "10px", right: "10px", width: "400px", height: "500px",
            background: "#000", border: "1px solid #0f0", color: "#0f0", fontFamily: "monospace",
            zIndex: "1000000", display: "flex", flexDirection: "column", opacity: "0.9"
        });
        
        div.innerHTML = `
            <div style="padding:5px; border-bottom:1px solid #0f0; display:flex; justify-content:space-between;">
                <b>NEXUS-2244</b> <span id="nexus-status">Idle</span> <button id="nexus-close">x</button>
            </div>
            <div id="nexus-graph" style="flex:1; background:#050505; position:relative;"></div>
            <div style="padding:5px; height:100px; overflow:auto; border-top:1px solid #0f0;" id="nexus-log"></div>
            <div style="padding:5px; display:flex;">
                <input id="nexus-input" type="text" style="flex:1; background:#111; color:#0f0; border:1px solid #0f0;" placeholder="Command NEXUS...">
                <button id="nexus-send">Send</button>
            </div>
            <div style="padding:5px; display:flex; justify-content:space-between;">
                 <button id="nexus-replay">Replay</button>
                 <button id="nexus-export">Export DB</button>
                 <button id="nexus-goals">Goals</button>
            </div>
        `;
        document.body.appendChild(div);

        document.getElementById("nexus-close").onclick = () => div.style.display = "none";
        document.getElementById("nexus-send").onclick = sendPrompt;
        document.getElementById("nexus-input").onkeydown = (e) => { if(e.key==="Enter") sendPrompt(); };
        document.getElementById("nexus-export").onclick = () => window.open(API+"/export");
        document.getElementById("nexus-replay").onclick = replayTimeline;

        initThreeJS(document.getElementById("nexus-graph"));
    }

    function log(msg) {
        const l = document.getElementById("nexus-log");
        if(l) {
            const d = document.createElement("div");
            d.textContent = "> " + msg;
            l.appendChild(d);
            l.scrollTop = l.scrollHeight;
        }
    }

    async function sendPrompt() {
        const input = document.getElementById("nexus-input");
        const prompt = input.value;
        if(!prompt) return;
        input.value = "";
        log("Thinking...");
        document.getElementById("nexus-status").textContent = "Reasoning...";

        const domTreeBefore = captureDOMTree();
        
        try {
            const r = await fetch(API + "/event", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    prompt: prompt,
                    dom: document.body.innerHTML, // Full HTML for analysis
                    diff: [] // Initial diff empty, backend handles memory diffs mostly
                })
            });
            const j = await r.json();
            log(`[${j.agent}] ${j.response}`);
            document.getElementById("nexus-status").textContent = "Idle";
            
            // Post-processing
            const domTreeAfter = captureDOMTree();
            const diff = diffDOMTrees(domTreeBefore, domTreeAfter);
            if(diff.length > 0) {
                log(`${diff.length} DOM changes`);
                renderDOMHeatmap(diff);
            }
            refreshGraph();

        } catch(e) {
            log("Error: " + e.message);
        }
    }

    async function replayTimeline() {
        log("Loading timeline...");
        const r = await fetch(API + "/timeline");
        const timeline = await r.json();
        log(`Found ${timeline.length} nodes. Replaying...`);
        // Simple visualization of replay
        timeline.forEach((node, i) => {
            setTimeout(() => {
                log(`[${new Date(node.time).toLocaleTimeString()}] ${node.agent || 'SYS'}`);
            }, i * 500);
        });
    }

    // --- THREE.JS GRAPH ---

    let scene, camera, renderer, meshes = [];

    function initThreeJS(container) {
        scene = new THREE.Scene();
        camera = new THREE.PerspectiveCamera(75, container.clientWidth / container.clientHeight, 0.1, 1000);
        camera.position.z = 30;
        
        renderer = new THREE.WebGLRenderer({ alpha: true });
        renderer.setSize(container.clientWidth, container.clientHeight);
        container.innerHTML = "";
        container.appendChild(renderer.domElement);

        const light = new THREE.PointLight(0xffffff, 1, 100);
        light.position.set(10, 10, 10);
        scene.add(light);
        
        animate();
        refreshGraph();
    }

    function animate() {
        requestAnimationFrame(animate);
        if(renderer && scene && camera) {
            meshes.forEach(m => m.rotation.y += 0.01);
            renderer.render(scene, camera);
        }
    }

    async function refreshGraph() {
        if(!scene) return;
        meshes.forEach(m => scene.remove(m));
        meshes = [];

        try {
            const r = await fetch(API + "/memory");
            const nodes = await r.json();
            
            nodes.slice(-50).forEach((n, i) => {
                const geo = new THREE.SphereGeometry(1, 16, 16);
                const color = n.branch === "main" ? 0x00ff00 : 0xff00ff;
                const mat = new THREE.MeshBasicMaterial({ color: color, wireframe: true });
                const mesh = new THREE.Mesh(geo, mat);
                
                // Spiral layout
                mesh.position.set(
                    Math.sin(i * 0.5) * 10,
                    Math.cos(i * 0.5) * 10,
                    i * -0.5 + 10
                );
                
                scene.add(mesh);
                meshes.push(mesh);
            });
        } catch(e) {
            console.error("Graph error", e);
        }
    }

    // --- INIT ---

    document.addEventListener("keydown", (e) => {
        if (e.ctrlKey && e.key === "Enter") {
            createUI();
        }
    });

    console.log("NEXUS-2244 Ultimate Bridge Loaded. Ctrl+Enter to Activate.");
})();
