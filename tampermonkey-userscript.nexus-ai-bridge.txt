    // Enhanced entropy system with SHA256/MD5 fractal rehashing
    class EntropyManager {
        constructor() {
            this.entropyBuffer = new Float64Array(256);
            this.entropyIndex = 0;
            this.qbitStates = new Map(); // For quantum-like entropy states
            this.initEntropyBuffer();
        }

        async initEntropyBuffer() {
            // Initialize with cryptographically secure pseudo-random values
            for (let i = 0; i < this.entropyBuffer.length; i++) {
                this.entropyBuffer[i] = Math.random();
            }
        }

        async sha256(data) {
            const encoder = new TextEncoder();
            const hashBuffer = await crypto.subtle.digest('SHA-256', encoder.encode(data));
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
        }

        async md5ChunkHash(data) {
            // Simple MD5-like hash for chunking (non-crypto, for ordering only)
            let hash = 0;
            for (let i = 0; i < data.length; i++) {
                hash = ((hash << 5) - hash) + data.charCodeAt(i);
                hash |= 0; // Convert to 32bit integer
            }
            return (hash >>> 0).toString(16); // Unsigned
        }

        async fractalRehash(baseHash, iterations = 3) {
            let currentHash = baseHash;
            for (let i = 0; i < iterations; i++) {
                const entropy = this.entropyBuffer[(this.entropyIndex++) % this.entropyBuffer.length];
                const shift = Math.floor(entropy * 256);
                const rotated = currentHash.substring(shift) + currentHash.substring(0, shift);
                currentHash = await this.sha256(rotated + entropy.toString());

                // Qbit-like superposition of entropy states
                this.qbitStates.set(`iteration_${i}`, {
                    hash: currentHash,
                    entropy: entropy,
                    timestamp: Date.now()
                });
            }
            return currentHash;
        }

        getShiftedEntropy(index) {
            const baseEntropy = this.entropyBuffer[index % this.entropyBuffer.length];
            const shift = (index * CONFIG.ENTROPY_MODULUS) % 1;
            return (baseEntropy + shift) % 1;
        }

        async generateGenesisHash(prompt, context) {
            const combined = prompt + JSON.stringify(context) + Date.now().toString();
            const sha256Hash = await this.sha256(combined);
            const md5Chunk = await this.md5ChunkHash(combined.substring(0, 1024));
            const fractalHash = await this.fractalRehash(sha256Hash);

            return {
                sha256: sha256Hash,
                md5Chunks: md5Chunk,
                fractal: fractalHash,
                genesis: await this.sha256(sha256Hash + md5Chunk + fractalHash),
                timestamp: Date.now(),
                entropy: this.getShiftedEntropy(Date.now() % 256)
            };
        }
    }

    class NexusMemoryGraph {
        constructor(bridge) {
            this.bridge = bridge;
            this.nodes = new Map();      // hash → node
            this.edges = new Set();      // parent → child
            this.scene = null;
            this.camera = null;
            this.renderer = null;
            this.meshes = new Map();
            this.selectedNode = null;
        }

        async loadMemories() {
            const memories = await this.bridge.getMemories(1000);
            for (const m of memories) {
                this.ingestMemory(m);
            }
        }

        ingestMemory(memory) {
            const g = memory.genesisHash?.genesis || memory.id;
            const node = {
                id: g,
                prompt: memory.prompt,
                entropy: memory.genesisHash?.entropy || 0,
                time: memory.timestamp,
                responses: memory.responses || [],
                best: memory.bestResponse,
                parent: memory.context?.genesisHash?.genesis || null
            };
            this.nodes.set(g, node);
            if (node.parent) this.edges.add([node.parent, g]);
        }

        /* ---------- Three.js ---------- */

        init3D(canvas) {
            this.scene = new THREE.Scene();
            this.camera = new THREE.PerspectiveCamera(75, canvas.clientWidth / canvas.clientHeight, 0.1, 1000);
            this.camera.position.z = 40;

            this.renderer = new THREE.WebGLRenderer({ canvas, alpha: true });
            this.renderer.setSize(canvas.clientWidth, canvas.clientHeight);

            const light = new THREE.DirectionalLight(0x00ff88, 1);
            light.position.set(1, 1, 1);
            this.scene.add(light);

            this.animate();
        }

        animate() {
            requestAnimationFrame(() => this.animate());
            this.renderer.render(this.scene, this.camera);
        }

        buildGraph() {
            this.meshes.forEach(m => this.scene.remove(m));
            this.meshes.clear();

            const layout = [...this.nodes.values()];
            layout.forEach((n, i) => {
                const geo = new THREE.SphereGeometry(1 + n.entropy * 3, 16, 16);
                const mat = new THREE.MeshBasicMaterial({
                    color: new THREE.Color().setHSL(0.35 - n.entropy * 0.35, 1, 0.5)
                });
                const mesh = new THREE.Mesh(geo, mat);
                mesh.position.set(
                    Math.sin(i) * 15,
                    Math.cos(i * 1.7) * 10,
                    i * -0.6
                );
                mesh.userData = n;
                this.scene.add(mesh);
                this.meshes.set(n.id, mesh);
            });

            // edges
            for (const [a, b] of this.edges) {
                const A = this.meshes.get(a);
                const B = this.meshes.get(b);
                if (!A || !B) continue;
                const g = new THREE.BufferGeometry().setFromPoints([A.position, B.position]);
                this.scene.add(new THREE.Line(g, new THREE.LineBasicMaterial({ color: 0x006600 })));
            }
        }

        pick(ray) {
            const hits = ray.intersectObjects([...this.meshes.values()]);
            if (!hits.length) return;
            this.inspect(hits[0].object.userData);
        }

        inspect(node) {
            this.selectedNode = node;
            alert(
                `GENESIS: ${node.id}\n` +
                `Entropy: ${node.entropy}\n` +
                `Prompt: ${node.prompt}\n` +
                `Responses: ${node.responses.length}\n` +
                `Time: ${new Date(node.time).toLocaleString()}`
            );
        }

        async replay() {
            const sorted = [...this.nodes.values()].sort((a, b) => a.time - b.time);
            for (const n of sorted) {
                console.log("⏳ REPLAY", n.prompt);
                await new Promise(r => setTimeout(r, 300));
            }
        }

        async export() {
            const blob = new Blob(
                [JSON.stringify([...this.nodes.values()], null, 2)],
                { type: "application/json" }
            );
            GM_download({ url: URL.createObjectURL(blob), name: "nexus_memory_graph.json" });
        }
    }

            GM_regerMenuCommand('Open NEXUS Dashboard', () => bridge.openNexusDashboard());
            GM_registerMenuCommand('Sync Current Page DOM', async () => {
                const dom = await bridge.captureFullDOM();
                alert(`DOM captured: ${dom.html.length} bytes`);
            });
            GM_registerMenuCommand('Export Memories', () => bridge.exportMemories());
            GM_registerMenuCommand('Check WSL Connection', async () => {
                try {
                    const result = await bridge.callWSLScript('Test connection');
                    alert(`WSL Connection: OK\\nResponse: ${result.substring(0, 200)}`);
                } catch (e) {
                    alert(`WSL Connection: FAILED\\nError: ${e.message}`);
                }
            });
        }

        return bridge;
    }

    // Start initialization
    let initialization = initializeNexus();

    // Provide async access to bridge
    window.getNexusBridge = async () => {
        return await initialization;
    };
