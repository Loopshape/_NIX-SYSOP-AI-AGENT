function generateHash(text) {
    const crypto = require('crypto');
    return {
        md5: crypto.createHash('md5').update(text).digest('hex'),
        sha256: crypto.createHash('sha256').update(text).digest('hex'),
        genesis: crypto.createHash('sha256')

// Calculate entropy (simple token-based)
function calculateEntropy(tokens) {
    if (!tokens || tokens.length === 0) return 0.0;
    
    const tokenCounts = {};
    tokens.forEach(token => {
        tokenCounts[token] = (tokenCounts[token] || 0) + 1;
    });
    
    const total = tokens.length;
    let entropy = 0;
    
    Object.values(tokenCounts).forEach(count => {
        const probability = count / total;
        entropy -= probability * Math.log2(probability);
    });
    
    return entropy;
}

        num_predict: 512,
        top_p: 0.9,

            prompt,
            stream: false,

            timeout: 30000
        });

        console.error('Ollama query failed:', error.message);
        
        // Fallback logic
        if (process.env.GEMINI_API_KEY) {
            console.log('Falling back to Gemini API');
            return await queryGemini(prompt);
        }
        
        throw new Error(`Ollama query failed: ${error.message}`);
    }
}

import fs from 'fs';
import crypto from 'crypto';

export class TimelineEngine {
    constructor(dbPath = 'memory/timeline.db') {
        this.dbPath = dbPath;
        this.ensureDatabase();
    }
    
    ensureDatabase() {
        if (!fs.existsSync(this.dbPath)) {
            fs.writeFileSync(this.dbPath, '');
        }
    }
    
    saveState(md5, data) {
        const entry = {
            md5,
            data,
            timestamp: Date.now(),
            hash: crypto.createHash('sha256')
                .update(JSON.stringify(data))
                .digest('hex')
        };
        
        fs.appendFileSync(
            this.dbPath,
            JSON.stringify(entry) + '\n'
        );
        
        return entry.hash;
    }
    
    replay(md5) {
        if (!fs.existsSync(this.dbPath)) {
            return [];
        }
        
        const content = fs.readFileSync(this.dbPath, 'utf8');
        return content
            .split('\n')
            .filter(line => line.trim())
            .map(line => JSON.parse(line))
            .filter(entry => entry.md5 === md5);
    }
    
    getBranch(md5) {
        if (!fs.existsSync(this.dbPath)) {
            return [];
        }
        
        const content = fs.readFileSync(this.dbPath, 'utf8');
        const entries = content
            .split('\n')
            .filter(line => line.trim())
            .map(line => JSON.parse(line));
        
        // Build tree structure
        const tree = [];
        const visited = new Set();
        
        function buildTree(currentMd5, depth = 0) {
            if (visited.has(currentMd5) || depth > 20) return [];
            
            visited.add(currentMd5);
            const children = entries.filter(e => e.data?.parent === currentMd5);
            
            return children.map(child => ({
                ...child,
                children: buildTree(child.md5, depth + 1)
            }));
        }
        
        return buildTree(md5);
    }
    
    calculateEntropy(md5) {
        const entries = this.replay(md5);
        if (entries.length === 0) return 0.0;
        
        // Calculate token diversity
        const allText = entries.map(e => e.data?.text || '').join(' ');
        const tokens = allText.split(/\s+/).filter(t => t.length > 0);
        
        if (tokens.length === 0) return 0.0;
        
        const tokenCounts = {};
        tokens.forEach(token => {
            tokenCounts[token] = (tokenCounts[token] || 0) + 1;
        });
        
        const total = tokens.length;
        let entropy = 0;
        
        Object.values(tokenCounts).forEach(count => {
            const probability = count / total;
            entropy -= probability * Math.log2(probability);
        });
        
        return entropy;
    }
}

#!/usr/bin/env node

import { SentenceTransformer } from 'sentence-transformers';
import sqlite3 from 'sqlite3';
import { open } from 'sqlite3';
import crypto from 'crypto';
import fs from 'fs';

async function main() {
    // Read text from stdin
    let text = '';
    process.stdin.setEncoding('utf8');
    
    for await (const chunk of process.stdin) {
        text += chunk;
    }
    
    if (!text.trim()) {
        console.error('No text provided to vectorize');
        process.exit(1);
    }
    
    // Generate hashes
    const md5 = crypto.createHash('md5').update(text).digest('hex');
    const sha256 = crypto.createHash('sha256').update(text).digest('hex');
    
    try {
        // Initialize sentence transformer
        const model = new SentenceTransformer('all-MiniLM-L6-v2');
        const embedding = await model.encode(text);
        
        // Convert to buffer for storage
        const vectorBuffer = Buffer.from(embedding);
        
        // Store in SQLite
        const db = await open({
            filename: 'memory/vectors.db',
            driver: sqlite3.Database
        });
        
        await db.run(`
            CREATE TABLE IF NOT EXISTS vectors (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                md5 TEXT UNIQUE NOT NULL,
                sha256 TEXT NOT NULL,
                vector BLOB NOT NULL,
                text TEXT NOT NULL,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
            )
        `);
        
        await db.run(
            `INSERT OR REPLACE INTO vectors (md5, sha256, vector, text) VALUES (?, ?, ?, ?)`,
            [md5, sha256, vectorBuffer, text]
        );
        
        await db.close();
        
        // Output result
        console.log(JSON.stringify({
            status: 'success',
            md5,
            sha256,
            vector_dimensions: embedding.length,
            text_preview: text.substring(0, 100) + '...'
        }));
        
    } catch (error) {
        console.error('Vectorization error:', error);
        console.log(JSON.stringify({
            status: 'error',
            error: error.message,
            md5,
            sha256
        }));
        process.exit(1);
    }
}

main().catch(console.error);

        cat "$stream_file" | node vector.js 2>/dev/null || true

        local token_count=$(wc -w < "$stream_file" 2>/dev/null || echo "0")
        local unique_tokens=$(tr ' ' '\n' < "$stream_file" | sort -u | wc -l 2>/dev/null || echo "0")
        local entropy=$(echo "scale=4; $unique_tokens / ($token_count + 1)" | bc 2>/dev/null || echo "0.0")
        
        # Store entropy
        echo "$entropy" > "entropy/${name}_${GENESIS}.entropy"
        
        log "INFO" "Agent $name completed: $token_count tokens, entropy: $entropy"
    fi
}

  private streamOllama(requestData: OllamaRequest): ReadableStream<OllamaResponse> {

        try {
          const response = await fetch(`${config.ollamaHost}/api/generate`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(requestData)
          });
          
          if (!response.body) {
            throw new Error('No response body from Ollama');
          }
          
          const reader = response.body.getReader();
          const decoder = new TextDecoder();
          let buffer = '';
          let fullResponse = '';
          
          while (true) {
            const { done, value } = await reader.read();
            
            if (done) {
              controller.enqueue({
                model: requestData.model,
                response: fullResponse,
                done: true
              });
              controller.close();
              break;
            }
            
            const chunk = decoder.decode(value, { stream: true });
            buffer += chunk;
            
            const lines = buffer.split('\n');
            buffer = lines.pop() || '';
            
            for (const line of lines) {
              if (line.trim()) {
                try {
                  const parsed = JSON.parse(line);
                  if (parsed.response) {
                    fullResponse += parsed.response;
                    controller.enqueue({
                      model: requestData.model,
                      response: parsed.response,
                      done: parsed.done || false
                    });
                  }
                } catch (e) {
                  console.warn('Failed to parse streaming chunk:', line);
                }
              }
            }
          }
        } catch (error) {
          controller.error(error);
        }
      }
    });

export class AgentResponseParser {
  static parseAgentResponse(
    rawResponse: string,
    expectedAgent: string,
    streaming: boolean = false,
    chunks: string[] = []
  ): AgentResponse {
    const strategies = [
      this.parseStructuredJson,
      this.parseMarkdownWithHeaders,
      this.parseKeyValueFormat,
      this.parseNaturalLanguage
    ];
    
    for (const strategy of strategies) {
      try {
        const parsed = strategy(rawResponse, expectedAgent);
        if (parsed && parsed.decision && parsed.reasoning) {
          return {
            ...parsed,
            rawResponse,
            streaming,
            chunks,
            confidence: parsed.confidence || this.calculateConfidence(parsed, chunks)
          };
        }
      } catch {
        continue;
      }
    }
    
    return this.createFallbackResponse(rawResponse, expectedAgent, streaming, chunks);
  }
  
  // Enhanced CODE agent prompt generator
  static generateCODEAgentPrompt(
    prompt: string,
    signInsights?: string,
    loopOptimizations?: string,
    corePrinciples?: string
  ): string {
    return `
[ROLE: CODE SYNTHESIZER - Interpreter & Executor]
[PRECEDENCE: CORE > CODE > LOOP > SIGN]

CONTEXT:
${corePrinciples ? `CORE PRINCIPLES:\n${corePrinciples}\n` : ''}
${signInsights ? `SIGN INTERPRETATIONS:\n${signInsights}\n` : 'Awaiting SIGN interpretation...'}
${loopOptimizations ? `LOOP OPTIMIZATIONS:\n${loopOptimizations}\n` : 'Awaiting LOOP optimization patterns...'}

INSTRUCTIONS:
1. Synthesize findings from all agents
2. Prioritize CORE principles when conflicts arise
3. Transform insights into executable code
4. Handle partial dependencies gracefully
5. Support fallback to Gemini if Ollama fails
6. Parse varied response formats

TASK: ${prompt}

RESPONSE FORMAT:
{
  "agent": "CODE",
  "decision": "Specific implementation approach",
  "reasoning": "Synthesis of all agent inputs",
  "confidence": 0.0-1.0,
  "dependencies": ["SIGN", "LOOP", "CORE"],
  "implementation": "Code snippet or pseudocode"
}

BEGIN SYNTHESIS:
`;
  }
  
  private static parseStructuredJson(
    rawResponse: string,
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    const jsonMatch = rawResponse.match(/\{[\s\S]*\}/);
    if (!jsonMatch) return null;
    
    try {
      const parsed = JSON.parse(jsonMatch[0]);
      return {
        agent: parsed.agent || parsed.role || expectedAgent,
        decision: parsed.decision || parsed.result || parsed.output || parsed.implementation || '',
        reasoning: parsed.reasoning || parsed.analysis || parsed.explanation || parsed.synthesis || '',
        confidence: parsed.confidence || parsed.certainty || 0.5
      };
    } catch {
      return null;
    }
  }
  
  private static parseMarkdownWithHeaders(
    rawResponse: string,
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    const decisionMatch = rawResponse.match(/##? (?:Decision|Implementation)[:\s]*\n*(.*?)(?=\n##? |$)/is);
    const reasoningMatch = rawResponse.match(/##? (?:Reasoning|Synthesis)[:\s]*\n*(.*?)(?=\n##? |$)/is);
    
    if (!decisionMatch && !reasoningMatch) return null;
    
    return {
      agent: expectedAgent,
      decision: decisionMatch?.[1]?.trim() || '',
      reasoning: reasoningMatch?.[1]?.trim() || '',
      confidence: this.extractConfidenceFromText(rawResponse)
    };
  }
  
  private static parseKeyValueFormat(
    rawResponse: string,
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    const lines = rawResponse.split('\n');
    let decision = '';
    let reasoning = '';
    
    for (const line of lines) {
      const lowerLine = line.toLowerCase();
      if (lowerLine.includes('decision:') || lowerLine.includes('implementation:')) {
        decision = line.split(':').slice(1).join(':').trim();
      } else if (lowerLine.includes('reasoning:') || lowerLine.includes('synthesis:')) {
        reasoning = line.split(':').slice(1).join(':').trim();
      }
    }
    
    if (!decision && !reasoning) return null;
    
    return {
      agent: expectedAgent,
      decision,
      reasoning,
      confidence: this.extractConfidenceFromText(rawResponse)
    };
  }
  
  private static parseNaturalLanguage(
    rawResponse: string,
    expectedAgent: string
  ): Partial<AgentResponse> | null {
    const sentences = rawResponse.split(/[.!?]+/).filter(s => s.trim());
    if (sentences.length === 0) return null;
    
    return {
      agent: expectedAgent,
      decision: sentences[0].trim(),
      reasoning: sentences.slice(1).join('. ').trim() || sentences[0].trim(),
      confidence: 0.3
    };
  }
  
  private static extractConfidenceFromText(text: string): number {
    const patterns = [
      /confidence[:\s]*(\d+\.?\d*)/i,
      /certainty[:\s]*(\d+\.?\d*)/i,
      /(\d+)% (confidence|certainty)/i,
      /I am (\d+)% sure/i
    ];
    
    for (const pattern of patterns) {
      const match = text.match(pattern);
      if (match) {
        const value = parseFloat(match[1]);
        if (!isNaN(value)) {
          return value > 1 ? value / 100 : value;
        }
      }
    }
    
    return 0.5;
  }
  
  private static calculateConfidence(parsed: Partial<AgentResponse>, chunks: string[]): number {
    let confidence = 0.5;
    
    // Longer reasoning suggests more confidence
    if (parsed.reasoning && parsed.reasoning.length > 100) confidence += 0.2;
    
    // Clear decision suggests more confidence
    if (parsed.decision && parsed.decision.length > 20) confidence += 0.2;
    
    // Multiple streaming chunks suggests thorough processing
    if (chunks.length > 3) confidence += 0.1;
    
    // Contains code or specific references
    if (parsed.decision?.match(/```|\b(def|function|class)\b/)) confidence += 0.1;
    
    return Math.min(0.95, confidence);
  }
  
  private static createFallbackResponse(
    rawResponse: string,
    expectedAgent: string,
    streaming: boolean,
    chunks: string[]
  ): AgentResponse {
    return {
      agent: expectedAgent,
      decision: 'Unable to parse specific decision',
      reasoning: streaming 
        ? `Streaming response assembled from ${chunks.length} chunks`
        : `Raw response: ${rawResponse.substring(0, 200)}...`,
      confidence: 0.1,
      rawResponse,
      streaming,
      chunks
    };
  }
}

            if (typeof window !== 'undefined') {

                  done: value.done
                }

            done: true
          };

          errors.push(`All fallbacks failed: ${fallbackError.message}`);

        }

      // Parse agent response
      const agentResponse = AgentResponseParser.parseAgentResponse(

  const [streamChunks, setStreamChunks] = useState<StreamingChunk[]>([]);
  const [activeStream, setActiveStream] = useState<string>('');
  const streamContainerRef = useRef<HTMLDivElement>(null);

  // Load models on mount
  useEffect(() => {
    loadModels();
    
    // Listen for streaming chunks
    const handleChunk = (event: CustomEvent) => {
      const chunk = event.detail as StreamingChunk;
      setStreamChunks(prev => [...prev, chunk]);
      
      if (chunk.done) {
        setStreaming(false);
        setActiveStream('');
      }
      
      // Auto-scroll
      if (streamContainerRef.current) {
        streamContainerRef.current.scrollTop = streamContainerRef.current.scrollHeight;
      }
    };
    
    window.addEventListener('ollama-chunk', handleChunk as EventListener);
    
    return () => {
      window.removeEventListener('ollama-chunk', handleChunk as EventListener);
    };
  }, []);

  const loadModels = async () => {
    try {
      const availableModels = await nexusEngine.getModels();
      setModels(availableModels);
      
      if (availableModels.length > 0) {
        const defaultModel = availableModels.find(m => m.name.includes('llama'))?.name || availableModels[0].name;
        setSelectedModel(defaultModel);
        nexusEngine.setActiveModel(defaultModel);
      }
    } catch (error) {
      console.error('Failed to load models:', error);
      setErrors(prev => [...prev, 'Failed to load models']);
    }
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    
    if (!prompt.trim()) return;
    
    setLoading(true);
    setResponse('');
    setErrors([]);
    setStreamChunks([]);
    
    // Update active model
    if (selectedModel) {
      nexusEngine.setActiveModel(selectedModel);
    }
    
    console.log(`[App] Processing prompt with model: ${selectedModel}, streaming: ${streaming}`);
    
    try {
      const result = await nexusEngine.processPrompt(prompt, {
        model: selectedModel,
        agent: 'CODE',
        stream: streaming,
        context: [{
          signInsights: 'Symbolic interpretation of patterns',
          loopOptimizations: 'Iterative refinement patterns',
          corePrinciples: 'Fundamental logical axioms'
        }]
      });
      
      setResponse(result.result);
      
      if (result.fallbackUsed) {
        console.warn(`[App] Fallback was triggered. Details:`, {
          originalModel: selectedModel,
          errors: result.errors,
          timestamp: new Date().toISOString()
        });
        
        setErrors(prev => [...prev, 'Fallback to Gemini was used']);
      }
      
      if (result.errors.length > 0) {
        setErrors(prev => [...prev, ...result.errors]);
      }
      
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : String(error);
      setErrors(prev => [...prev, `Processing failed: ${errorMessage}`]);
      console.error('[App] Processing failed:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleModelSelect = (modelName: string) => {
    setSelectedModel(modelName);
    const model = models.find(m => m.name === modelName);
    setModelDetails(model || null);
  };

  // Assemble streaming chunks into readable text
  const streamingText = streamChunks
    .filter(chunk => chunk.agent === activeStream)
    .map(chunk => chunk.chunk)
    .join('');

  return (
    <div className="min-h-screen bg-gray-900 text-gray-100 p-6">
      <div className="max-w-6xl mx-auto">
        <header className="mb-8">
          <h1 className="text-3xl font-bold text-green-400">NEXUS AI Orchestrator</h1>
          <p className="text-gray-400 mt-2">Multi-Agent System with Streaming Support</p>
        </header>

        <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
          {/* Left Panel - Controls */}
          <div className="lg:col-span-1 space-y-6">
            {/* Model Selector */}
            <div className="bg-gray-800 rounded-lg p-4">
              <h3 className="text-lg font-semibold mb-3">🧠 Ollama Models</h3>
              
              <div className="space-y-3">
                <select
                  value={selectedModel}
                  onChange={(e) => handleModelSelect(e.target.value)}
                  className="w-full bg-gray-700 border border-gray-600 rounded px-3 py-2 text-sm"
                  disabled={loading}
                >
                  <option value="">Select a model...</option>
                  {models.map(model => (
                    <option key={model.name} value={model.name}>
                      {model.name}
                    </option>
                  ))}
                </select>
                
                {modelDetails && (
                  <div className="bg-gray-900 rounded p-3 text-sm space-y-2">
                    <div className="flex justify-between">
                      <span className="text-gray-400">Family:</span>
                      <span>{modelDetails.details.family || 'Unknown'}</span>
                    </div>
                    <div className="flex justify-between">
                      <span className="text-gray-400">Digest:</span>
                      <span className="font-mono text-xs">
                        {modelDetails.digest.substring(0, 16)}...
                      </span>
                    </div>
                    {modelDetails.details.parameter_size && (
                      <div className="flex justify-between">
                        <span className="text-gray-400">Parameters:</span>
                        <span>{modelDetails.details.parameter_size}</span>
                      </div>
                    )}
                    {modelDetails.details.quantization && (
                      <div className="flex justify-between">
                        <span className="text-gray-400">Quantization:</span>
                        <span>{modelDetails.details.quantization}</span>
                      </div>
                    )}
                    {modelDetails.details.modified_at && (
                      <div className="flex justify-between">
                        <span className="text-gray-400">Modified:</span>
                        <span>
                          {new Date(modelDetails.details.modified_at).toLocaleDateString()}
                        </span>
                      </div>
                    )}
                  </div>
                )}
              </div>
              
              <div className="mt-4 pt-4 border-t border-gray-700">
                <label className="flex items-center space-x-2">
                  <input
                    type="checkbox"
                    checked={streaming}
                    onChange={(e) => setStreaming(e.target.checked)}
                    className="rounded"
                  />
                  <span>Enable Streaming</span>
                </label>
              </div>
            </div>
            
            {/* Status */}
            <div className="bg-gray-800 rounded-lg p-4">
              <h3 className="text-lg font-semibold mb-3">📊 Status</h3>
              <div className="space-y-2">
                <div className="flex justify-between">
                  <span className="text-gray-400">Active Model:</span>
                  <span className="font-medium">{selectedModel || 'None'}</span>
                </div>
                <div className="flex justify-between">
                  <span className="text-gray-400">Streaming:</span>
                  <span className={streaming ? 'text-green-400' : 'text-yellow-400'}>
                    {streaming ? 'Enabled' : 'Disabled'}
                  </span>
                </div>
                <div className="flex justify-between">
                  <span className="text-gray-400">Available Models:</span>
                  <span>{models.length}</span>
                </div>
              </div>
            </div>
          </div>
          
          {/* Main Panel - Input & Output */}
          <div className="lg:col-span-2 space-y-6">
            {/* Input Form */}
            <div className="bg-gray-800 rounded-lg p-4">
              <form onSubmit={handleSubmit} className="space-y-4">
                <div>
                  <label className="block text-sm font-medium text-gray-300 mb-2">
                    Enter Prompt for CODE Agent
                  </label>
                  <textarea
                    value={prompt}
                    onChange={(e) => setPrompt(e.target.value)}
                    rows={4}
                    className="w-full bg-gray-700 border border-gray-600 rounded px-3 py-2 text-sm"
                    placeholder="Describe what you want the CODE agent to implement..."
                    disabled={loading}
                  />
                  <p className="text-xs text-gray-500 mt-1">
                    The CODE agent will synthesize insights from SIGN, LOOP, and CORE agents
                  </p>
                </div>
                
                <div className="flex justify-between items-center">
                  <button
                    type="submit"
                    disabled={loading || !prompt.trim()}
                    className="px-6 py-2 bg-green-600 text-white rounded hover:bg-green-700 disabled:opacity-50"
                  >
                    {loading ? 'Processing...' : 'Send to CODE Agent'}
                  </button>
                  
                  <button
                    type="button"
                    onClick={() => nexusEngine.simulateAndTestFallback()}
                    className="px-4 py-2 bg-yellow-600 text-white rounded hover:bg-yellow-700 text-sm"
                  >
                    Test Fallback
                  </button>
                </div>
              </form>
            </div>
            
            {/* Streaming Output */}
            {streaming && streamChunks.length > 0 && (
              <div className="bg-gray-800 rounded-lg p-4">
                <h3 className="text-lg font-semibold mb-3">⚡ Live Stream</h3>
                <div
                  ref={streamContainerRef}
                  className="bg-gray-900 rounded p-4 h-64 overflow-y-auto font-mono text-sm"
                >
                  {streamingText.split('').map((char, index) => (
                    <span key={index} className="text-green-300">
                      {char}
                    </span>
                  ))}
                  {streaming && !streamChunks[streamChunks.length - 1]?.done && (
                    <span className="animate-pulse">▊</span>
                  )}
                </div>
              </div>
            )}
            
            {/* Response */}
            {response && (
              <div className="bg-gray-800 rounded-lg p-4">
                <h3 className="text-lg font-semibold mb-3">🤖 Response</h3>
                <div className="bg-gray-900 rounded p-4 whitespace-pre-wrap font-mono text-sm">
                  {response}
                </div>
              </div>
            )}
            
            {/* Errors */}
            {errors.length > 0 && (
              <div className="bg-red-900/30 border border-red-700 rounded-lg p-4">
                <h3 className="text-lg font-semibold text-red-300 mb-2">⚠️ Errors</h3>
                <ul className="space-y-1">
                  {errors.map((error, index) => (
                    <li key={index} className="text-sm text-red-200">
                      • {error}
                    </li>
                  ))}
                </ul>
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  );
};

export default App;
