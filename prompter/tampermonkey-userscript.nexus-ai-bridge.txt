(function() {
    'use strict';

    const CONFIG = {
        OLLAMA_HOST: 'http://localhost:11434',
        OLLAMA_WSL_PATH: '/home/loop/_/ai/ai.sh',
        NEXUS_API: 'http://localhost:3000',
        AGENT_POOL: ["CORE", "CUBE", "LOOP", "SIGN", "LINE", "COIN", "WORK", "CODE"],
        ENTROPY_MODULUS: Math.PI * 2 / 8,
        REQUEST_TIMEOUT: 30000,
        MEMORY_CHUNK_SIZE: 16384,
        MAX_TOKENS: 4096
    };

    // -------------------- Entropy Manager --------------------
    class EntropyManager {
        constructor() {
            this.entropyBuffer = new Float64Array(256);
            this.entropyIndex = 0;
            this.qbitStates = new Map();
            this.initEntropyBuffer();
        }

        async initEntropyBuffer() {
            for (let i = 0; i < this.entropyBuffer.length; i++) {
                this.entropyBuffer[i] = Math.random();
            }
        }

        async sha256(data) {
            const encoder = new TextEncoder();
            const hashBuffer = await crypto.subtle.digest('SHA-256', encoder.encode(data));
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            return hashArray.map(b => b.toString(16).padStart(2,'0')).join('');
        }

        async md5ChunkHash(data) {
            let hash = 0;
            for (let i = 0; i < data.length; i++) {
                hash = ((hash << 5) - hash) + data.charCodeAt(i);
                hash |= 0;
            }
            return (hash >>> 0).toString(16);
        }

        async fractalRehash(baseHash, iterations=3) {
            let currentHash = baseHash;
            for (let i=0;i<iterations;i++){
                const entropy = this.entropyBuffer[(this.entropyIndex++) % this.entropyBuffer.length];
                const shift = Math.floor(entropy*256);
                const rotated = currentHash.substring(shift) + currentHash.substring(0,shift);
                currentHash = await this.sha256(rotated + entropy.toString());
                this.qbitStates.set(`iteration_${i}`, {hash: currentHash, entropy: entropy, timestamp: Date.now()});
            }
            return currentHash;
        }

        getShiftedEntropy(index) {
            const baseEntropy = this.entropyBuffer[index % this.entropyBuffer.length];
            const shift = (index*CONFIG.ENTROPY_MODULUS) % 1;
            return (baseEntropy + shift) % 1;
        }

        async generateGenesisHash(prompt, context) {
            const combined = prompt + JSON.stringify(context) + Date.now().toString();
            const sha256Hash = await this.sha256(combined);
            const md5Chunk = await this.md5ChunkHash(combined.substring(0,1024));
            const fractalHash = await this.fractalRehash(sha256Hash);
            return {
                sha256: sha256Hash,
                md5Chunks: md5Chunk,
                fractal: fractalHash,
                genesis: await this.sha256(sha256Hash + md5Chunk + fractalHash),
                timestamp: Date.now(),
                entropy: this.getShiftedEntropy(Date.now()%256)
            };
        }
    }

    // -------------------- WSL Bridge --------------------
    class WSLBridge {
        constructor() {
            this.entropyManager = new EntropyManager();
            this.activeConnections = new Map();
            this.messageQueue = [];
            this.isProcessing = false;
            this.pinnedRepos = null;
        }

        async init() {
            await this.loadPinnedRepos();
            this.setupHotkey();
            this.setupMessageHandlers();
            return this;
        }

        async loadPinnedRepos() {
            try {
                const response = await fetch('https://api.github.com/users/Loopshape/repos');
                const repos = await response.json();
                this.pinnedRepos = repos.filter(r => r.topics && r.topics.includes('pinned')).map(r=>({
                    name: r.name, description: r.description, url: r.html_url, language: r.language, stars: r.stargazers_count
                }));
                console.log('Loaded pinned repos:', this.pinnedRepos.length);
            } catch(e){ this.pinnedRepos = []; }
        }

        setupHotkey() {
            document.addEventListener('keydown', (e) => {
                if (e.ctrlKey && !e.altKey && e.key === 'Enter') {
                    e.preventDefault();
                    this.openNexusDashboard();
                }
            });
        }

        async openNexusDashboard() {
            const dashboardURL = chrome.runtime?.getURL('nexus-dashboard.html') ||
                  `data:text/html,${encodeURIComponent(this.generateDashboardHTML())}`;
            window.open(dashboardURL,'_blank');
        }

        generateDashboardHTML() {
            return `
<!DOCTYPE html>
<html>
<head><title>NEXUS AIP v9</title></head>
<body>
<canvas id="nexus-graph" style="width:100%;height:100vh;"></canvas>
<script>${this.getDashboardScript()}</script>
</body>
</html>`;
        }

        getDashboardScript() {
            return `
${NEXUS_DASHBOARD_SCRIPT.toString()}
window.nexusDashboard = new NexusDashboard();`;
        }

        setupMessageHandlers() {
            window.addEventListener('message', (event) => {
                if(event.data.type==='NEXUS_PROMPT'){
                    this.processPrompt(event.data.prompt,event.data.model);
                }
            });
        }

        async processPrompt(prompt, model='llama2'){
            const domState = { html: document.documentElement.outerHTML, timestamp: Date.now() };
            const genesisHash = await this.entropyManager.generateGenesisHash(prompt, domState);
            // Multi-agent concurrent processing
            const results = await Promise.all(CONFIG.AGENT_POOL.map(async (agent,i)=>{
                const agentEntropy = this.entropyManager.getShiftedEntropy(i);
                const agentPrompt = \`[AGENT:\${agent}] [ENTROPY:\${agentEntropy.toFixed(6)}] [GENESIS_HASH:\${genesisHash.genesis}] [USER_PROMPT:\${prompt}]\`;
                try { return await this.callWSLScript(agentPrompt); } catch(e){ return '[TIMEOUT]'; }
            }));
            // Sort by entropy
            const validResults = results.map((r,i)=>({agent: CONFIG.AGENT_POOL[i], response:r, entropy:this.entropyManager.getShiftedEntropy(i)}))
                                        .sort((a,b)=>a.entropy-b.entropy);
            const best = validResults[0];
            best.finalHash = await this.entropyManager.fractalRehash(genesisHash.genesis + best.response);
            await this.storeMemory({prompt, responses:validResults, bestResponse:best, genesisHash, timestamp:Date.now()});
            return best;
        }

        async callWSLScript(command){
            return new Promise((resolve,reject)=>{
                GM_xmlhttpRequest({
                    method:'POST',
                    url: CONFIG.OLLAMA_HOST+'/api/generate',
                    headers:{'Content-Type':'application/json'},
                    data: JSON.stringify({model:'llama2',prompt:command,stream:false}),
                    timeout: CONFIG.REQUEST_TIMEOUT,
                    onload: r=>{try{resolve(JSON.parse(r.responseText).response);}catch(e){resolve(r.responseText);}},
                    onerror: reject,
                    ontimeout: reject
                });
            });
        }

        async storeMemory(memItem){
            const memories = GM_getValue('nexus_memories',[]);
            memories.push({...memItem,id:await this.entropyManager.sha256(JSON.stringify(memItem)+Date.now())});
            if(memories.length>100) memories.splice(0, memories.length-100);
            GM_setValue('nexus_memories',memories);
        }

        async getMemories(limit=50){
            return GM_getValue('nexus_memories',[]).slice(-limit);
        }

        async exportMemories(){
            const mem = await this.getMemories(1000);
            const blob = new Blob([JSON.stringify(mem,null,2)],{type:'application/json'});
            GM_download({url:URL.createObjectURL(blob),name:'nexus_memories_'+Date.now()+'.json',saveAs:true});
        }
    }

    // -------------------- Dashboard Script --------------------
function NEXUS_DASHBOARD_SCRIPT(){
    class NexusDashboard{
        constructor(){ this.scene=null; this.camera=null; this.renderer=null; this.memoryNodes=[]; this.timelineSlider=null; this.currentMemoryIndex=0; this.heatmapOverlay=null; this.initThreeJSMemoryGraph(); this.initTimelineScrubber(); this.initHeatmapOverlay(); }
        initHeatmapOverlay(){ const o=document.createElement('div');o.id='nexus-heatmap';o.style.position='fixed';o.style.top='0';o.style.left='0';o.style.width='100%';o.style.height='100%';o.style.pointerEvents='none';o.style.zIndex='999999';o.style.background='rgba(255,0,0,0)';document.body.appendChild(o);this.heatmapOverlay=o;}
        applyHeatmap(mem){ if(!mem||!mem.domDiff)return; let max=mem.domDiff.length; let html=''; mem.domDiff.forEach((d,i)=>{const a=((i+1)/max)*0.6; html+=\`<div style="position:absolute;top:\${d.top}px;left:\${d.left}px;width:\${d.width}px;height:\${d.height}px;background:rgba(255,0,0,\${a});pointer-events:none;"></div>\`;}); this.heatmapOverlay.innerHTML=html;}
            initTimelineScrubber(){ const s=document.createElement('input');s.type='range';s.min=0;s.max=0;s.value=0;s.style.position='fixed';s.style.bottom='10px';s.style.left='50px';s.style.width='calc(100%-100px)';s.style.zIndex='999999';document.body.appendChild(s); this.timelineSlider=s; s.addEventListener('input',async ()=>{ const idx=parseInt(s.value); await this.scrubMemoryTo(idx); }); }
            async scrubMemoryTo(idx){ const mems=await window.NEXUS_BRIDGE.getMemories(100); if(idx>=mems.length)idx=mems.length-1; this.currentMemoryIndex=idx; this.timelineSlider.max=mems.length-1; this.highlightNode(mems[idx]); this.applyHeatmap(mems[idx]); }
            highlightNode(mem){ this.memoryNodes.forEach(n=>n.material.color.set(0x0f0)); const n=this.memoryNodes.find(x=>x.userData.memory.id===mem.id); if(n)n.material.color.set(0xff0); }
            async replayBranch(mem){ if(!mem.branch||mem.branch.length===0)return; for(let i=0;i<mem.branch.length;i++){ const m=mem.branch[i]; this.highlightNode(m); this.applyHeatmap(m); await new Promise(r=>setTimeout(r,600)); } }
            async initThreeJSMemoryGraph(){ const canvas=document.getElementById('nexus-graph'); this.scene=new THREE.Scene(); this.camera=new THREE.PerspectiveCamera(75,canvas.clientWidth/canvas.clientHeight,0.1,1000); this.renderer=new THREE.WebGLRenderer({canvas,antialias:true}); this.renderer.setSize(canvas.clientWidth,canvas.clientHeight); this.renderer.setClearColor(0x0b0f14,1); this.scene.add(new THREE.AmbientLight(0x404040)); const dl=new THREE.DirectionalLight(0x00ff00,0.5); dl.position.set(1,1,1); this.scene.add(dl); this.camera.position.z=15; this.memoryNodes=[]; await this.loadMemoryNodes(); this.raycaster=new THREE.Raycaster(); this.mouse=new THREE.Vector2(); canvas.addEventListener('click',async e=>{const r=canvas.getBoundingClientRect();this.mouse.x=((e.clientX-r.left)/r.width)*2-1;this.mouse.y=-((e.clientY-r.top)/r.height)*2+1;this.raycaster.setFromCamera(this.mouse,this.camera); const hits=this.raycaster.intersectObjects(this.memoryNodes); if(hits.length>0){ const mem=hits[0].object.userData.memory; await this.scrubMemoryTo(this.memoryNodes.indexOf(hits[0].object)); await this.replayBranch(mem); } }); this.animateGraph(); }
            async loadMemoryNodes(){ const mems=await window.NEXUS_BRIDGE.getMemories(100); this.memoryNodes.forEach(n=>this.scene.remove(n)); this.memoryNodes=[]; mems.forEach((mem,i)=>{ const geo=new THREE.SphereGeometry(0.2,16,16); const mat=new THREE.MeshPhongMaterial({color:0x0f0}); const s=new THREE.Mesh(geo,mat); const a=i*0.3,r=3+i*0.05; s.position.set(r*Math.cos(a),i*0.1,r*Math.sin(a)); s.userData.memory=mem; this.scene.add(s); this.memoryNodes.push(s); }); }
            animateGraph(){ requestAnimationFrame(()=>this.animateGraph()); this.memoryNodes.forEach(n=>{ n.rotation.x+=0.005; n.rotation.y+=0.01; }); this.renderer.render(this.scene,this.camera); }
        }
    }

    // -------------------- Initialize NEXUS --------------------

        console.log('🔄 Initializing NEXUS AIP v9...');
        const bridge = new WSLBridge();
        await bridge.init();
        window.NEXUS_BRIDGE = bridge;

        const hint=document.createElement('div'); hint.className='nexus-shortcut-hint'; hint.textContent='NEXUS: CTRL+ENTER'; hint.title='Open NEXUS AI Dashboard'; document.body.appendChild(hint);
        if(typeof GM_registerMenuCommand!=='undefined'){
            GM_registerMenuCommand('Open NEXUS Dashboard',()=>bridge.openNexusDashboard());
            GM_registerMenuCommand('Export Memories',()=>bridge.exportMemories());
        }
        return bridge;
    }

    let initialization=initializeNexus();
    window.getNexusBridge=async()=>await initialization;

})();
