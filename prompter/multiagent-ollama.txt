# Send a prompt to Ollama and retrieve response
ollama_query() {
    local model="$1"
    local prompt="$2"
    local max_tokens="${3:-256}"
    local temperature="${4:-0.5}"
    
    # Basic payload
    local payload
    payload=$(jq -nc --arg model "$model" \
                   --arg prompt "$prompt" \
                   --argjson max_tokens "$max_tokens" \
                   --argjson temperature "$temperature" \
                   '{
                        model: $model,
                        prompt: $prompt,
                        options: {num_predict: $max_tokens, temperature: $temperature}
                     }')

    # Call Ollama API
    local response
    response=$(curl -s -X POST "$OLLAMA_HOST/api/generate" \
        -H "Content-Type: application/json" \
        -d "$payload" --max-time 10)

    # Error handling
    if [[ $? -ne 0 ]] || [[ -z "$response" ]]; then
        echo "[ERROR] Ollama API call failed or timed out"
        return 1
    fi

    # Parse response safely
    local text
    text=$(jq -r '.response // empty' <<<"$response")
    if [[ -z "$text" ]]; then
        echo "[WARN] Ollama returned empty response, fallback simulated"
        text="(SIMULATED) Ollama response unavailable"
    fi
    echo "$text"
}

# Fetch available models from Ollama
ollama_list_models() {
    local models
    models=$(curl -s "$OLLAMA_HOST/api/models" --max-time 5 | jq -r '.[].name')
    if [[ -z "$models" ]]; then
        echo "[WARN] Could not retrieve model list from Ollama"
        echo "default_model"
    else
        echo "$models"
    fi
}

execute_agent_interaction() {
    local agent_id="$1"
    local prompt="$2"
    declare -n previous_outputs="$3" # associative array of other agents outputs
    local agent_name="${AGENTS[$agent_id]}"

    # Build combined context from other agents
    local combined_context=""
    for other_id in "${!previous_outputs[@]}"; do
        combined_context+="[Agent ${AGENTS[$other_id]} Output]: ${previous_outputs[$other_id]}\n"
    done

    local full_prompt="${combined_context}\nTask for ${agent_name}:\n${prompt}"

    # Choose model (example: default, can be dynamic via UI)
    local model="${SELECTED_MODEL:-default_model}"

    # Query Ollama
    local result
    result=$(ollama_query "$model" "$full_prompt" 512 0.5)

    # Store result for other agents
    previous_outputs["$agent_id"]="$result"

    echo "$result"
}

    # Ensure at least 1 and not more than base
    (( optimal < 1 )) && optimal=1

    echo "$optimal"
}

estimate_tokens() {
    local text="$1"
    # Rough estimation: ~4 chars per token for English text
    local chars=${#text}
    echo $(( (chars + 3) / 4 ))
}

truncate_prompt() {
    local prompt="$1"
    local max_chars="${2:-$MAX_PROMPT_CHARS}"
    
    if [[ ${#prompt} -le $max_chars ]]; then
        echo "$prompt"
        return
    fi
    
    # Smart truncation: try to cut at sentence boundaries
    local truncated
    truncated="${prompt:0:$max_chars}"
    
    # Find last sentence boundary
    local last_period last_newline last_boundary
    last_period=$(echo "$truncated" | grep -o '[.!?]' | tail -1 | cat)
    last_newline=$(echo "$truncated" | tail -c 200 | grep -o $'\n' | tail -1 | cat)
    
    if [[ -n "$last_period" ]]; then
        # Cut at last complete sentence
        truncated="${truncated%$last_period*}$last_period"
    elif [[ -n "$last_newline" ]]; then
        # Cut at last paragraph
        truncated="${truncated%$last_newline*}"
    else:
        # Force cut and add ellipsis
        truncated="${truncated:0:$((max_chars - 3))}..."
    fi
    
    local original_tokens estimated_tokens
    original_tokens=$(estimate_tokens "$prompt")
    estimated_tokens=$(estimate_tokens "$truncated")
    
    log_message "WARN" "Prompt truncated: ${original_tokens} → ${estimated_tokens} tokens"
    echo "$truncated"
}

_ollama_post_with_retry() {
    local endpoint="$1" payload="$2" max_retries="${3:-$DEFAULT_RETRY_ATTEMPTS}"
    local attempt=0 delay=1 max_delay=30
    
    while (( attempt <= max_retries )); do
        local resp http_code
        resp=$(curl -s -w "\n%{http_code}" -X POST \
            "${OLLAMA_HOST}${endpoint}" \
            -H "Content-Type: application/json" \
            -d "$payload" \
            --max-time 120 \
            --connect-timeout 30)
        
        http_code=$(echo "$resp" | tail -1)
        resp=$(echo "$resp" | sed '$d')
        
        if [[ "$http_code" == "200" ]]; then
            echo "$resp"
            return 0
        fi
        
        ((attempt++))
        if (( attempt > max_retries )); then
            log_message "ERROR" "Ollama request failed after $max_retries attempts"
            return 1
        fi
        
        # Exponential backoff with jitter
        local jitter
        jitter=$(awk "BEGIN {printf \"%.2f\", 0.5 + (RANDOM % 1000)/1000}")
        sleep_seconds=$(awk "BEGIN {printf \"%.2f\", $delay * $jitter}")
        
        log_message "WARN" "Attempt $attempt failed (HTTP $http_code), retrying in ${sleep_seconds}s..."
        sleep "$sleep_seconds"
        ((delay = delay * 2 > max_delay ? max_delay : delay * 2))
    done
}

resolve_input() {
    local input="$1"
    local cache_dir="${BASE_DIR}/cache"
    mkdir -p "$cache_dir"
    
    # Check cache first (with 1-hour TTL)
    local cache_key
    cache_key=$(echo -n "$input" | sha256sum | cut -d' ' -f1)
    local cache_file="${cache_dir}/${cache_key}"
    
    if [[ -f "$cache_file" ]]; then
        local cache_age
        cache_age=$(( $(date +%s) - $(stat -c %Y "$cache_file") ))
        if (( cache_age < 3600 )); then
            log_message "DEBUG" "Cache hit for input (${cache_age}s old)"
            cat "$cache_file"
            return
        fi
    fi
    
    # Original resolution logic
    local content=""
    if [[ "$input" =~ ^https?:// ]]; then
        log_message "INFO" "Input detected as URL: $input"
        content=$(curl -sL --max-time 30 "$input" 2>/dev/null || echo "[URL fetch failed]")
        content=$(echo "$content" | sed 's/<[^>]*>//g' | tr -s ' \t\n' ' ')
    elif [[ -f "$input" ]]; then
        log_message "INFO" "Input detected as file: $input"
        # Limit file size for safety
        local file_size
        file_size=$(stat -c%s "$input" 2>/dev/null || echo "0")
        if (( file_size > 1048576 )); then  # 1MB
            log_message "WARN" "Large file ($((file_size/1024))KB), reading first 1MB only"
            content=$(head -c 1048576 "$input")
        else
            content=$(cat "$input")
        fi
    elif [[ "$input" =~ ^[a-fA-F0-9]{64}$ ]]; then
        log_message "INFO" "Input detected as Hash: $input"
        content=$(sql_exec "SELECT response FROM agent_memories WHERE prompt LIKE '%$input%' LIMIT 1;")
        [[ -z "$content" ]] && content="Hash reference found but no content"
    else
        content="$input"
    fi
    
    # Truncate if necessary
    content=$(truncate_prompt "$content")
    
    # Cache the result
    echo "$content" > "$cache_file"
    echo "$content"
}

        log_message "WARN" "Large prompt: ~${prompt_tokens} tokens (model context may be exceeded)"
    fi
    
    local temp_dir

    trap 'rm -rf "$temp_dir"' EXIT INT TERM
    
    log_message "INFO" "Starting 8-agent cycle for: ${prompt:0:50}..."
    
    # Update system state
    sql_exec "UPDATE system_state SET value='${#AGENTS[@]}' WHERE key='active_agents';"

    local dynamic_concurrency
    dynamic_concurrency=$(calculate_optimal_concurrency "$MAX_CONCURRENT_AGENTS")

                '.agents[] | select(.id == $id or .name == "'${AGENTS[$agent_id]}'")' 2>/dev/null || echo "")
        fi
        
        # Wait for available slot based on dynamic concurrency
        while (( $(jobs -r | wc -l) >= dynamic_concurrency )); do
            # Recalculate concurrency in case system load changed
            dynamic_concurrency=$(calculate_optimal_concurrency "$MAX_CONCURRENT_AGENTS")
            sleep 0.1
        done
        
        # Execute agent
        (
            local result
            result=$(execute_agent_with_telemetry "$agent_id" "$prompt" "$agent_override" "$dry_run")
            echo "$result" > "$temp_dir/agent_${agent_id}.json"
        ) &
        pids+=($!)
    done
    
    # Wait for all agents with progress indication
    local total_agents=${#AGENTS[@]}
    local completed=0
    
    while (( completed < total_agents )); do
        wait -n
        ((completed++))
        
        # Update progress every 2 agents
        if (( completed % 2 == 0 )) || (( completed == total_agents )); then
            log_message "DEBUG" "Progress: $completed/$total_agents agents completed"
        fi
        
        # Recalculate concurrency periodically
        if (( completed % 4 == 0 )); then
            dynamic_concurrency=$(calculate_optimal_concurrency "$MAX_CONCURRENT_AGENTS")
        fi
    done
    
    # Collect results
    for agent_id in "${!AGENTS[@]}"; do
        if [[ -s "$temp_dir/agent_${agent_id}.json" ]]; then
            agent_results+=("$(cat "$temp_dir/agent_${agent_id}.json")")
        else
            agent_results+=("{\"agent_id\":$agent_id,\"agent_name\":\"${AGENTS[$agent_id]}\",\"response\":\"[FAILED]\",\"error\":\"No output generated\"}")
        fi
    done
    
    # Coordinator phase (unless dry-run)
    local coord_resp="[DRY-RUN] Coordinator summary skipped"
    if [[ "$dry_run" != "true" ]]; then

                '{model:$model, prompt:$prompt, stream:false}')
        
        if _check_ollama; then
            local coord_raw
            coord_raw=$(_ollama_post_with_retry "/api/generate" "$payload")
            coord_resp=$(echo "$coord_raw" | jq -r '.response // empty')
        fi
    fi
    
    # Update cycle phase
    local new_phase
    new_phase=$(( ( $(sql_exec "SELECT value FROM system_state WHERE key='cycle_phase';") + 1 ) % ${#AGENTS[@]} ))
    sql_exec "UPDATE system_state SET value='$new_phase' WHERE key='cycle_phase';"
    sql_exec "UPDATE system_state SET value='0' WHERE key='active_agents';"
    
    # Cleanup
    rm -rf "$temp_dir"
    
    # Final output
    jq -n \
        --arg prompt "$prompt" \
        --argjson phase "$new_phase" \
        --argjson workers "$(printf "%s\n" "${agent_results[@]}" | jq -s '.')" \
        --arg coordinator "$coord_resp" \
        --argjson dry_run "$dry_run" \
        '{
            status: "success",
            prompt: $prompt,
            cycle_phase: $phase,
            workers: $workers,
            coordinator: $coordinator,
            dry_run: $dry_run
        }'
}

process_batch() {
    local batch_file="$1" dry_run="${2:-$DRY_RUN}"
    
    if [[ ! -f "$batch_file" ]]; then
        log_message "ERROR" "Batch file not found: $batch_file"
        return 1
    fi
    
    log_message "INFO" "Processing batch file: $batch_file"
    
    # Load or create state
    local state
    state=$(jq -n '{
        batch_file: $batch,
        start_time: now|strftime("%Y-%m-%d %H:%M:%S"),
        processed: 0,
        failed: 0,
        current_index: 0,
        items: []
    }' --arg batch "$batch_file")
    
    echo "$state" > "$STATE_FILE"
    
    # JSON Mode
    if [[ "$batch_file" =~ \.json$ ]]; then
        local count
        count=$(jq '. | length' "$batch_file")
        log_message "INFO" "Found $count items in JSON batch"
        
        for ((i=0; i<count; i++)); do
            # Update state
            state=$(jq --argjson i "$i" '.current_index = $i' "$STATE_FILE")
            echo "$state" > "$STATE_FILE"
            
            local item
            item=$(jq -c ".[${i}]" "$batch_file")
            
            local item_prompt
            item_prompt=$(jq -r '.prompt' <<<"$item")
            
            # Extract overrides
            local overrides=""
            if jq -e '.overrides' <<<"$item" >/dev/null 2>&1; then
                overrides=$(jq -c '.overrides' <<<"$item")
            fi
            
            log_message "INFO" "Processing item $((i+1))/$count: ${item_prompt:0:50}..."
            
            local result
            if result=$(parallel_reasoning "$item_prompt" "$overrides" "$dry_run" 2>/dev/null); then
                state=$(jq --argjson i "$i" '.processed += 1' "$STATE_FILE")
                state=$(jq --argjson i "$i" --arg result "$result" '.items[$i] = ($result | fromjson)' <<<"$state")
            else
                state=$(jq --argjson i "$i" '.failed += 1' "$STATE_FILE")
                state=$(jq --argjson i "$i" --arg error "Processing failed" '.items[$i] = {error: $error}' <<<"$state")
                log_message "ERROR" "Failed to process item $((i+1))"
            fi
            
            echo "$state" > "$STATE_FILE"
            
            # Progress update
            if (( (i + 1) % 5 == 0 )) || (( i + 1 == count )); then
                local processed failed
                processed=$(jq '.processed' "$STATE_FILE")
                failed=$(jq '.failed' "$STATE_FILE")
                log_message "INFO" "Batch progress: $processed/$count processed, $failed failed"
            fi
            
            # Small delay to prevent overwhelming the system
            sleep 0.1
        done
        
        log_message "INFO" "Batch complete: $count items processed"
        
        # Generate batch summary
        local summary
        summary=$(jq -n \
            --argfile state "$STATE_FILE" \
            '{
                batch_file: $state.batch_file,
                start_time: $state.start_time,
                end_time: now|strftime("%Y-%m-%d %H:%M:%S"),
                total_items: ($state.items | length),
                processed: $state.processed,
                failed: $state.failed,
                items: $state.items
            }')
        
        echo "$summary" | jq .
        return 0
    else
        # Text mode (simpler)
        local i=0
        while IFS= read -r line || [[ -n "$line" ]]; do
            ((i++))
            [[ -z "$line" ]] && continue
            
            log_message "INFO" "Processing line $i: ${line:0:50}..."
            parallel_reasoning "$line" "" "$dry_run"
            
            if (( i % 10 == 0 )); then
                log_message "INFO" "Progress: $i lines processed"
            fi
            
            sleep 0.05
        done < "$batch_file"
    fi
}

                break
                ;;

    shift || true
    
    init_database
    
    # Handle resume
    if [[ "$resume" == "true" ]]; then
        resume_batch
        exit $?
    fi
    
    case "$cmd" in
        reason)
            [[ $# -lt 1 ]] && { echo "Usage: $0 [--dry-run] reason \"<prompt>\""; exit 1; }
            parallel_reasoning "$1" "${2:-}" "$dry_run"
            ;;
        
        batch)

            ;;
        
        status)
            system_status
            ;;
        
        telemetry)
            echo "=== Structured Telemetry ==="
            [[ -f "$STRUCTURED_LOG" ]] && tail -20 "$STRUCTURED_LOG" | jq .
            echo "=== Recent Logs ==="
            tail -10 "$LOG_FILE"
            ;;
        
        preview)
            [[ $# -lt 2 ]] && { echo "Usage: $0 preview <command> <args>"; exit 1; }
            DRY_RUN=true
            export DRY_RUN
            case "$1" in
                reason) parallel_reasoning "$2" "${3:-}" "true" ;;
                batch) process_batch "$2" "true" ;;

            esac
            ;;

            log_message "INFO" "Optimizing database and cleaning caches..."
            sql_exec "VACUUM;"
            sql_exec "ANALYZE;"
            rm -rf "${BASE_DIR}/cache"/*
            log_message "INFO" "Optimization complete"
            ;;
        
        help|*)
            show_help
            ;;
    esac
}

export interface OllamaOptions {
    model: string;
    temperature?: number;
    num_predict?: number;
    stream?: boolean; // streaming flag
}

export interface AgentResponse {
    agent: string;
    decision: string;
    raw?: any;
}

export async function queryOllama(prompt: string, options: OllamaOtions): Promise<AgentResponse[]> {
    const url = `http://localhost:11434/completions`;
    try {
        const response = await axios.post(
            url,
            {
                model: options.model,
                prompt,
                temperature: options.temperature ?? 0.7,
                num_predict: options.num_predict ?? 1,
                stream: options.stream ?? false
            },
            { timeout: 8000 } // 8 sec timeout
        );

        // Handle streaming or normal response
        const data = response.data;
        let parsed: AgentResponse[] = [];

        if (options.stream && Array.isArray(data)) {
            // streaming: data may come as chunks
            for (const chunk of data) {
                parsed.push(parseOllamaChunk(chunk));
            }
        } else {
            parsed = parseOllamaResponse(data);
        }

        return parsed;

    } catch (err) {
        handleOllamaError(err);
        // Fallback to Gemini
        console.warn('[FALLBACKOllama failed, switching to Gemini API...');
        return await queryGemini(prompt); // assume queryGemini exists
    }
}

// ---------------- Parsing Helpers ----------------

function parseOllamaChunk(chunk: any): AgentResponse {
    try {
        if (typeof chunk === 'string') {
            return { agent: 'unknown', decision: chunk, raw: chunk };
        }
        if (chunk.agent && chunk.decision) {
            return { agent: chunk.agent, decision: chunk.decision, raw: chunk };
        }
        return { agent: 'unknown', decision: JSON.stringify(chunk), raw: chunk };
    } catch (e) {
        console.error('[OLLAMA PARSE CHUNK ERROR]', e, chunk);
        return { agent: 'unknown', decision: '[parse error]', raw: chunk };
    }
}

function parseOllamaResponse(data: any): AgentResponse[] {
    try {
        if (Array.isArray(data)) {
            return data.map(d => parseOllamaChunk(d));
        }
        return [parseOllamaChunk(data)];
    } catch (e) {
        console.error('[OLLAMA PARSE RESPONSE ERROR]', e, data);
        return [{ agent: 'unknown', decision: '[parse error]', raw: data }];
    }
}

// ---------------- Error Handling ----------------

function handleOllamaError(err: unknown) {
    if (axios.isAxiosError(err)) {
        const axiosErr = err as AxiosError;
        if (axiosErr.code === 'ECONNREFUSED') console.error('[OLLAMA ERROR] Connection refused');
        else if (axiosErr.code === 'ECONNABORTED') console.error('[OLLAMA ERROR] Timeout');
        else console.error('[OLLAMA ERROR]', axiosErr.message, axiosErr.response?.data);
    } else {
        console.error('[OLLAMA ERROR]', err);
    }
}

  temperature?: number;
  num_predict?: number;

  frequency_penalty?: number;
  stop?: string[];
  stream?: boolean;

// Streaming handler callback
export type StreamCallback = (chunk: string) => void;

// Main query function for Ollama
export async function queryOllama(
  prompt: string,
  model: string,
  options: OllamaOptions = {},
  onChunk?: StreamCallback
): Promise<any> {
  const payload: any = {
    model,
    prompt,
    ...options
  };

  try {
    const response = await fetch('http://localhost:11434/v1/completions', {
      method: 'POST',
      body: JSON.stringify(payload),
      headers: { 'Content-Type': 'application/json' },
      timeout: options.timeout || 90000
    });

    if (!response.ok) throw new Error(`Ollama API error: ${response.statusText}`);

    if (options.stream && onChunk) {
      // Stream handling (pseudo, depends on actual Ollama streaming API)
      for await (const chunk of response.body) {
        const text = chunk.toString();
        onChunk(text);
      }
      return null;
    } else {
      const result = await response.json();
      return parseAgentResponse(result);
    }
  } catch (err: any) {
    console.error('Ollama error, falling back to Gemini:', err.message);
    return queryGemini(prompt, options, onChunk); // seamless fallback
  }
}

MEMORY_FILE = "agent_memory.json"

def load_memory():
    if os.path.exists(MEMORY_FILE):
        with open(MEMORY_FILE, 'r') as f:
            return json.load(f)
    return {}

def save_memory(memory):
    with open(MEMORY_FILE, 'w') as f:
        json.dump(memory, f, indent=2)

def index_prompt(prompt: str, result: str):
    memory = load_memory()
    timestamp = time.time()
    sha256 = hashlib.sha256(prompt.encode()).hexdigest()
    md5 = hashlib.md5(result.encode()).hexdigest()

    # Entropy-based queue
    memory[sha256] = {
        "prompt": prompt,
        "result": result,
        "timestamp": timestamp,
        "md5_backtrace": md5,
        "steps": memory.get(sha256, {}).get("steps", 0) + 1
    }
    save_memory(memory)
    return sha256

export function initMindmap(container: HTMLElement, nodes: any[]) {
  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(75, container.clientWidth/container.clientHeight, 0.1, 1000);
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setSize(container.clientWidth, container.clientHeight);
  container.appendChild(renderer.domElement);

  const sphereGeometry = new THREE.SphereGeometry(0.5, 32, 32);
  const material = new THREE.MeshStandardMaterial({ color: 0x00ff00 });

  nodes.forEach((n, idx) => {
    const sphere = new THREE.Mesh(sphereGeometry, material);
    sphere.position.set(Math.sin(idx)*2, Math.cos(idx)*2, idx*0.5);
    scene.add(sphere);
    gsap.to(sphere.position, { y: sphere.position.y + 0.5, repeat: -1, yoyo: true, duration: 1 + idx*0.1 });
  });

  camera.position.z = 10;
  const light = new THREE.DirectionalLight(0xffffff, 1);
  light.position.set(5,5,5);
  scene.add(light);

  function animate() {
    requestAnimationFrame(animate);
    renderer.render(scene, camera);
  }
  animate();
}

        <label>Ollama Model:</label>
        <select value={activeModel} onChange={e => setActiveModel(e.target.value)}>
          {modelList.map(model => <option key={model} value={model}>{model}</option>)}
        </select>

        <textarea
          value={prompt}
          onChange={e => setPrompt(e.target.value)}
          placeholder="Enter your prompt..."
        />

        <button onClick={handlePromptSubmit}>Send</button>

        <div className="streaming-output">
          {streamingChunks.map((chunk, idx) => <div key={idx} className="chunk">{chunk}</div>)}
        </div>
      </div>

      <div id="mindmap-container" style={{ width: '100%', height: '400px' }}></div>
    </div>
  );
}

import fetch from 'node-fetch';

export interface OllamaOptions {
  temperature?: number;
  num_predict?: number;
  top_p?: number;
  presence_penalty?: number;
  frequency_penalty?: number;
  stop?: string[];
  stream?: boolean;
  timeout?: number;
}

export type StreamCallback = (chunk: string) => void;

export async function queryOllama(
  prompt: string,
  model: string,
  options: OllamaOptions = {},
  onChunk?: StreamCallback
): Promise<any> {
  const payload: any = { model, prompt, ...options };
  try {
    const response = await fetch('http://localhost:11434/v1/completions', {
      method: 'POST',
      body: JSON.stringify(payload),
      headers: { 'Content-Type': 'application/json' },
      timeout: options.timeout || 90000
    });

    if (!response.ok) throw new Error(`Ollama API error: ${response.statusText}`);

    if (options.stream && onChunk && response.body) {
      for await (const chunk of response.body) onChunk(chunk.toString());
      return null;
    } else {
      const result = await response.json();
      return parseAgentResponse(result);
    }
  } catch (err: any) {
    console.error('Ollama error, falling back to Gemini:', err.message);
    return queryGemini(prompt, options, onChunk);
  }
}

export async function processPrompt(prompt: string, model: string, onChunk?: StreamCallback) {
  const CODE_AGENT_PROMPT = `
    You are the CODE agent. Synthesize findings from SIGN and LOOP.
    Precedence: CORE > CODE > LOOP > SIGN.
    Finalize output only after all dependencies respond.
    If partial responses exist, state "Awaiting dependency sync".
    Use Gemini fallback if Ollama unavailable.
  `;
  const finalPrompt = `${CODE_AGENT_PROMPT}\nUser prompt: ${prompt}`;
  return queryOllama(finalPrompt, model, { stream: !!onChunk, temperature: 0.7, num_predict: 1 }, onChunk);
}

async function queryGemini(prompt: string, options: OllamaOptions, onChunk?: StreamCallback) {
  const simulated = `Gemini fallback response for: ${prompt}`;
  if (onChunk) onChunk(simulated);
  return simulated;
}

function parseAgentResponse(response: any) {
  try {
    if (response?.result) return response.result;
    return response;
  } catch {
    return { error: 'Unable to parse agent response' };
  }
}

import json, hashlib, os, time

MEMORY_FILE = "agent_memory.json"

def load_memory():
    if os.path.exists(MEMORY_FILE):
        with open(MEMORY_FILE, 'r') as f:
            return json.load(f)
    return {}

def save_memory(memory):
    with open(MEMORY_FILE, 'w') as f:
        json.dump(memory, f, indent=2)

def index_prompt(prompt: str, result: str):
    memory = load_memory()
    timestamp = time.time()
    sha256 = hashlib.sha256(prompt.encode()).hexdigest()
    md5 = hashlib.md5(result.encode()).hexdigest()

    memory[sha256] = {
        "prompt": prompt,
        "result": result,
        "timestamp": timestamp,
        "md5_backtrace": md5,
        "steps": memory.get(sha256, {}).get("steps", 0) + 1
    }
    save_memory(memory)
    return sha256

import * as THREE from 'three';
import gsap from 'gsap';

export function initMindmap(container: HTMLElement, nodes: any[]) {
  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(75, container.clientWidth/container.clientHeight, 0.1, 1000);
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setSize(container.clientWidth, container.clientHeight);
  container.appendChild(renderer.domElement);

  const material = new THREE.MeshStandardMaterial({ color: 0x00ff00 });
  const sphereGeometry = new THREE.SphereGeometry(0.5, 32, 32);

  nodes.forEach((n, idx) => {
    const sphere = new THREE.Mesh(sphereGeometry, material);
    sphere.position.set(Math.sin(idx)*2, Math.cos(idx)*2, idx*0.5);
    scene.add(sphere);
    gsap.to(sphere.position, { y: sphere.position.y + 0.5, repeat: -1, yoyo: true, duration: 1 + idx*0.1 });
  });

  camera.position.z = 10;
  const light = new THREE.DirectionalLight(0xffffff, 1);
  light.position.set(5,5,5);
  scene.add(light);

  function animate() {
    requestAnimationFrame(animate);
    renderer.render(scene, camera);
  }
  animate();
}
