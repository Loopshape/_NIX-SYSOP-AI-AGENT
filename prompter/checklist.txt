class NexusMemoryGraph {
    constructor(bridge) {
        this.bridge = bridge;
        this.nodes = new Map();      // hash → node
        this.edges = new Set();      // parent → child
        this.scene = null;
        this.camera = null;
        this.renderer = null;
        this.meshes = new Map();
        this.selectedNode = null;
    }

    async loadMemories() {
        const memories = await this.bridge.getMemories(1000);
        for (const m of memories) {
            this.ingestMemory(m);
        }
    }

    ingestMemory(memory) {
        const g = memory.genesisHash?.genesis || memory.id;
        const node = {
            id: g,
            prompt: memory.prompt,
            entropy: memory.genesisHash?.entropy || 0,
            time: memory.timestamp,
            responses: memory.responses || [],
            best: memory.bestResponse,
            parent: memory.context?.genesisHash?.genesis || null
        };
        this.nodes.set(g, node);
        if (node.parent) this.edges.add([node.parent, g]);
    }

    /* ---------- Three.js ---------- */

    init3D(canvas) {
        this.scene = new THREE.Scene();
        this.camera = new THREE.PerspectiveCamera(75, canvas.clientWidth / canvas.clientHeight, 0.1, 1000);
        this.camera.position.z = 40;

        this.renderer = new THREE.WebGLRenderer({ canvas, alpha: true });
        this.renderer.setSize(canvas.clientWidth, canvas.clientHeight);

        const light = new THREE.DirectionalLight(0x00ff88, 1);
        light.position.set(1, 1, 1);
        this.scene.add(light);

        this.animate();
    }

    animate() {
        requestAnimationFrame(() => this.animate());
        this.renderer.render(this.scene, this.camera);
    }

    buildGraph() {
        this.meshes.forEach(m => this.scene.remove(m));
        this.meshes.clear();

        const layout = [...this.nodes.values()];
        layout.forEach((n, i) => {
            const geo = new THREE.SphereGeometry(1 + n.entropy * 3, 16, 16);
            const mat = new THREE.MeshBasicMaterial({
                color: new THREE.Color().setHSL(0.35 - n.entropy * 0.35, 1, 0.5)
            });
            const mesh = new THREE.Mesh(geo, mat);
            mesh.position.set(
                Math.sin(i) * 15,
                Math.cos(i * 1.7) * 10,
                i * -0.6
            );
            mesh.userData = n;
            this.scene.add(mesh);
            this.meshes.set(n.id, mesh);
        });

        // edges
        for (const [a, b] of this.edges) {
            const A = this.meshes.get(a);
            const B = this.meshes.get(b);
            if (!A || !B) continue;
            const g = new THREE.BufferGeometry().setFromPoints([A.position, B.position]);
            this.scene.add(new THREE.Line(g, new THREE.LineBasicMaterial({ color: 0x006600 })));
        }
    }

    pick(ray) {
        const hits = ray.intersectObjects([...this.meshes.values()]);
        if (!hits.length) return;
        this.inspect(hits[0].object.userData);
    }

    inspect(node) {
        this.selectedNode = node;
        alert(
            `GENESIS: ${node.id}\n` +
            `Entropy: ${node.entropy}\n` +
            `Prompt: ${node.prompt}\n` +
            `Responses: ${node.responses.length}\n` +
            `Time: ${new Date(node.time).toLocaleString()}`
        );
    }

    async replay() {
        const sorted = [...this.nodes.values()].sort((a, b) => a.time - b.time);
        for (const n of sorted) {
            console.log("⏳ REPLAY", n.prompt);
            await new Promise(r => setTimeout(r, 300));
        }
    }

    async export() {
        const blob = new Blob(
            [JSON.stringify([...this.nodes.values()], null, 2)],
            { type: "application/json" }
        );
        GM_download({ url: URL.createObjectURL(blob), name: "nexus_memory_graph.json" });
    }
}

captureDOMTree() {
    function walk(node, path = "0") {
        let out = [];
        if (!node || !node.tagName) return out;

        const id = node.id || "";
        const cls = node.className || "";
        const sig = node.tagName + "#" + id + "." + cls;

        out.push({ path, sig });

        Array.from(node.children).forEach((child, i) => {
            out.push(...walk(child, path + "." + i));
        });

        return out;
    }

    return walk(document.body);
}

2️⃣ Compute per-node deltas

Add:

diffDOMTrees(before, after) {
    const mapA = new Map(before.map(n => [n.path, n.sig]));
    const mapB = new Map(after.map(n => [n.path, n.sig]));

    const changes = [];

    for (const [path, sig] of mapB.entries()) {
        if (!mapA.has(path)) {
            changes.push({ path, type: "added", sig });
        } else if (mapA.get(path) !== sig) {
            changes.push({ path, type: "modified", from: mapA.get(path), to: sig });
        }
    }

    for (const [path, sig] of mapA.entries()) {
        if (!mapB.has(path)) {
            changes.push({ path, type: "removed", sig });
        }
    }

    return changes;
}

import express from "express";
import bodyParser from "body-parser";
import Database from "better-sqlite3";
import crypto from "crypto";
import fetch from "node-fetch";

const app = express();
app.use(bodyParser.json({limit:"50mb"}));

const db = new Database("nexus.db");

db.exec(`
CREATE TABLE IF NOT EXISTS memory (
  id TEXT PRIMARY KEY,
  parent TEXT,
  time INTEGER,
  agent TEXT,
  prompt TEXT,
  response TEXT,
  domdiff TEXT,
  entropy REAL
);
`);

function hash(x){
  return crypto.createHash("sha256").update(x).digest("hex");
}

async function ollama(prompt){
  const r = await fetch("http://localhost:11434/api/generate",{
    method:"POST",
    headers:{ "Content-Type":"application/json" },
    body: JSON.stringify({ model:"gemma3:1b", prompt, stream:false })
  });
  return (await r.json()).response;
}

let lastNode = null;

app.post("/event", async (req,res)=>{
  const {dom, diff, prompt} = req.body;

  const response = await ollama(
    `You are NEXUS. Analyze DOM and diff.\nDOM:\n${dom}\nDIFF:\n${JSON.stringify(diff)}\nUser:${prompt}`
  );

  const entropy = diff.length + response.length;
  const id = hash(Date.now()+response);

  db.prepare(`INSERT INTO memory VALUES (?,?,?,?,?,?,?,?)`)
    .run(id,lastNode,Date.now(),"CORE",prompt,response,JSON.stringify(diff),entropy);

  lastNode = id;

  res.json({response,id});
});

app.get("/memory", (req,res)=>{
  res.json(db.prepare("SELECT * FROM memory").all());
});

app.listen(3000, ()=>console.log("NEXUS online on :3000"));

🧬 1️⃣ Multi-Agent Brain (Server Upgrade)

Replace your Ollama call in server.js with a swarm:

const models = ["gemma3:1b","mistral","neural-chat"];

async function swarm(prompt){
  const votes = [];
  for(const m of models){
    const r = await fetch("http://localhost:11434/api/generate",{
      method:"POST",
      headers:{ "Content-Type":"application/json" },
      body: JSON.stringify({ model:m, prompt, stream:false })
    });
    const j = await r.json();
    votes.push({model:m, text:j.response});
  }

  // consensus
  return votes.sort((a,b)=>b.text.length-a.text.length)[0];
}
