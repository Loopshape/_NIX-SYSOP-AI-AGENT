PHASE 3 — WebSocket Router
-----------------------------------------------------------

Write server.js:

import WebSocket, { WebSocketServer } from "ws";

const wss = new WebSocketServer({ port: 2244 });
let clients = [];

wss.on("connection", ws => {
  clients.push(ws);
  ws.on("close", () => clients = clients.filter(c => c !== ws));
});

export function broadcast(agent, token, sha256, genesis) {
  const msg = JSON.stringify({ agent, token, sha256, genesis });
  clients.forEach(c => c.send(msg));
}

PHASE 4 — ai.sh (NEXUS Kernel)
-----------------------------------------------------------

Write ai.sh:

#!/usr/bin/env bash
set -e

mkdir -p entropy memory

GENESIS=$(date +%s%N | sha256sum | cut -d' ' -f1)
PROMPT="$*"
echo "{ \"genesis\": \"$GENESIS\", \"states\": {} }" > memory/memory.json

declare -A AGENTS
AGENTS[coder]="deepseek-coder"
AGENTS[agi]="deepseek-r1"
AGENTS[nemodian]="wave"
AGENTS[sysop]="llama3.1"

stream_agent() {
  name=$1
  model=$2
  ollama run "$model" "$PROMPT" --stream | while read -r token; do
    echo "$token" >> "entropy/$name.stream"
    sha=$(echo -n "$token" | sha256sum | cut -d' ' -f1)
    echo "$sha" >> "entropy/$name.sha"
    node -e "
      import {broadcast} from './server.js';
      broadcast('$name', '$token', '$sha', '$GENESIS');
    "
  done
}

for a in "${!AGENTS[@]}"; do
  stream_agent "$a" "${AGENTS[$a]}" &
done
wait
